/// State for the @context / +project autocomplete popup (D-09).
#[derive(Debug, Clone)]
pub struct AutocompleteState {
    /// The trigger character: '@' or '+'.
    pub trigger: char,
    /// Current prefix text typed after the trigger (lowercase for matching).
    pub prefix: String,
    /// Filtered list of completion candidates (without trigger char).
    pub items: Vec<String>,
    /// Index of the currently highlighted item in the popup.
    pub selected: usize,
    /// Whether the popup has keyboard focus (Down arrow was pressed).
    pub focused: bool,
}

impl AutocompleteState {
    pub fn new(trigger: char, prefix: String, items: Vec<String>) -> Self {
        AutocompleteState { trigger, prefix, items, selected: 0, focused: false }
    }
}

**Step 3 — Token collection helper:**

Add a method to `App` that collects all unique tokens for a given trigger char from the current task list:

```rust
fn collect_tokens(&self, trigger: char) -> Vec<String> {
    let mut tokens: Vec<String> = self.task_list.tasks().iter().flat_map(|t| {
        if trigger == '@' {
            t.contexts.clone()
        } else {
            t.projects.clone()
        }
    }).collect();
    tokens.sort();
    tokens.dedup();
    tokens
}
```

Note: `Task.contexts` contains context tags WITHOUT the `@` prefix; `Task.projects` contains project tags WITHOUT `+`. The popup will display them without the trigger char, and insert `{trigger}{token}` when accepted.

**Step 4 — Trigger detection helper:**

After passing a key to the editor via `input_without_shortcuts()`, inspect the current editor content to detect whether autocomplete should activate/update:

```rust
fn update_autocomplete(&mut self) {
    // Only active in Adding or Editing mode.
    match self.mode {
        AppMode::Adding | AppMode::Editing { .. } => {}
        _ => { self.autocomplete = None; return; }
    }

    let line = self.editor.lines().first().cloned().unwrap_or_default();
    // Find the last occurrence of '@' or '+' in the line.
    // The prefix is everything from that char to the end of the line.
    let trigger_pos = line.rfind(|c| c == '@' || c == '+');

    if let Some(pos) = trigger_pos {
        let trigger = line.chars().nth(pos).unwrap();
        let prefix = &line[pos + 1..]; // text after the trigger char
        // Only show popup if prefix contains no spaces (i.e., still in the token).
        if prefix.contains(' ') {
            self.autocomplete = None;
            return;
        }
        let prefix_lower = prefix.to_lowercase();
        let all_tokens = self.collect_tokens(trigger);
        let filtered: Vec<String> = all_tokens.into_iter()
            .filter(|t| t.to_lowercase().starts_with(&prefix_lower))
            .collect();

        if filtered.is_empty() {
            self.autocomplete = None;
            return;
        }

        // Update existing state (preserve focused/selected if trigger+prefix unchanged).
        if let Some(ref mut ac) = self.autocomplete {
            if ac.trigger == trigger && ac.prefix == prefix.to_string() {
                // Same state — update items in case task list changed.
                ac.items = filtered;
                ac.selected = ac.selected.min(ac.items.len().saturating_sub(1));
                return;
            }
        }
        // New autocomplete state.
        self.autocomplete = Some(AutocompleteState::new(trigger, prefix.to_string(), filtered));
    } else {
        self.autocomplete = None;
    }
}
```

**Step 5 — Wire into Adding/Editing key dispatch:**

In the `AppMode::Adding | AppMode::Editing { .. }` match arm, add autocomplete handling BEFORE passing keys to the editor. The interaction stack rule: "autocomplete popup > editor" (PITFALLS.md).

```rust
AppMode::Adding | AppMode::Editing { .. } => {
    match key.code {
        KeyCode::Esc => {
            if self.autocomplete.is_some() {
                // Close popup only; keep editor open (D-08).
                self.autocomplete = None;
            } else {
                // No popup — cancel edit entirely.
                self.exit_edit_mode();
            }
        }
        KeyCode::Enter => {
            if let Some(ref ac) = self.autocomplete.clone() {
                if ac.focused {
                    // Popup is focused — Enter accepts the selected completion.
                    self.accept_completion();
                } else {
                    // Popup visible but not focused — Enter saves the edit.
                    self.autocomplete = None;
                    // ... (same save logic as before)
                    self.save_and_exit();
                }
            } else {
                // No popup — save the edit.
                self.save_and_exit();
            }
        }
        // Down arrow — move focus into the popup if it is visible (D-08).
        KeyCode::Down => {
            if let Some(ref mut ac) = self.autocomplete {
                ac.focused = true;
                ac.selected = (ac.selected + 1).min(ac.items.len().saturating_sub(1));
            } else {
                // No popup — pass Down to the editor (no-op in single-line mode).
                use crossterm::event::Event;
                self.editor.input_without_shortcuts(Event::Key(key));
            }
        }
        // Up arrow — move selection up in focused popup.
        KeyCode::Up => {
            if let Some(ref mut ac) = self.autocomplete {
                if ac.focused {
                    ac.selected = ac.selected.saturating_sub(1);
                }
            } else {
                use crossterm::event::Event;
                self.editor.input_without_shortcuts(Event::Key(key));
            }
        }
        // Tab or Space — accept selected completion if popup is focused (D-08).
        KeyCode::Tab | KeyCode::Char(' ') => {
            if self.autocomplete.as_ref().map(|ac| ac.focused).unwrap_or(false) {
                self.accept_completion();
                // For Space: also insert a space after the completion.
                if key.code == KeyCode::Char(' ') {
                    use crossterm::event::Event;
                    self.editor.input_without_shortcuts(Event::Key(key));
                }
            } else {
                // Pass to editor normally.
                use crossterm::event::Event;
                self.editor.input_without_shortcuts(Event::Key(key));
                self.update_autocomplete();
            }
        }
        // All other keys — pass to editor, then re-check autocomplete trigger.
        _ => {
            use crossterm::event::Event;
            self.editor.input_without_shortcuts(Event::Key(key));
            self.update_autocomplete();
        }
    }
}
```

**Step 6 — `accept_completion()` and `save_and_exit()` helpers:**

```rust
/// Insert the currently selected autocomplete token into the editor.
fn accept_completion(&mut self) {
    if let Some(ref ac) = self.autocomplete.clone() {
        if let Some(token) = ac.items.get(ac.selected) {
            // Replace the current prefix with the full token.
            // The editor content is: "...{trigger}{prefix}" → replace prefix with token.
            let line = self.editor.lines().first().cloned().unwrap_or_default();
            // Find last trigger char position.
            if let Some(pos) = line.rfind(|c: char| c == ac.trigger) {
                // Delete everything from pos+1 to end, then insert token.
                let new_line = format!("{}{}{}", &line[..=pos], token, "");
                // Rebuild editor with new content.
                let mut new_editor = tui_textarea::TextArea::default();
                new_editor.insert_str(&new_line);
                // Move cursor to end of new content.
                self.editor = new_editor;
            }
        }
    }
    self.autocomplete = None;
}

/// Save editor content and exit edit mode (used by Enter handler).
fn save_and_exit(&mut self) {
    let raw = self.editor.lines().first().cloned().unwrap_or_default();
    let raw = raw.trim().to_string();
    if !raw.is_empty() {
        let task = todotxt_core::Task::parse(&raw);
        match self.mode.clone() {
            AppMode::Adding => {
                if let Err(e) = self.task_list.add(task) {
                    eprintln!("add error: {e}");
                } else {
                    self.selected = self.task_list.len().saturating_sub(1);
                }
            }
            AppMode::Editing { original_idx } => {
                if let Err(e) = self.task_list.update(original_idx, task) {
                    eprintln!("edit error: {e}");
                }
            }
            _ => {}
        }
    }
    self.exit_edit_mode();
}
```

**Popup placement calculation:**

The popup appears directly above the footer row, anchored to the left edge of the frame. Height is capped at min(items.len(), 5) to avoid consuming the whole screen.

```rust
/// Render the autocomplete popup if active (D-08).
/// Call this AFTER rendering the task list and footer — overlays on top.
fn render_autocomplete_popup(&self, frame: &mut Frame, footer_area: ratatui::layout::Rect) {
    use ratatui::layout::Rect;
    use ratatui::style::{Color, Modifier, Style};
    use ratatui::widgets::{Block, Borders, List, ListItem, ListState};

    let ac = match &self.autocomplete {
        Some(ac) if !ac.items.is_empty() => ac,
        _ => return,
    };

    // Popup height: at most 5 rows, at least 1.
    let popup_height = (ac.items.len() as u16).min(5).min(footer_area.y);
    if popup_height == 0 {
        return; // Not enough room above footer.
    }

    // Position: just above the footer row, left-aligned.
    let popup_width = ac.items.iter()
        .map(|s| s.len() + 3) // +3 for trigger char + borders
        .max()
        .unwrap_or(20)
        .min(40) as u16;
    // Clamp to frame width.
    let popup_width = popup_width.min(frame.area().width);

    let popup_area = Rect {
        x: footer_area.x,
        y: footer_area.y.saturating_sub(popup_height),
        width: popup_width,
        height: popup_height,
    };

    // Build list items — prefix each with the trigger char.
    let items: Vec<ListItem> = ac.items.iter().map(|token| {
        ListItem::new(format!("{}{}", ac.trigger, token))
    }).collect();

    let highlight_style = if ac.focused {
        Style::default().add_modifier(Modifier::REVERSED)
    } else {
        Style::default().add_modifier(Modifier::DIM)
    };

    let popup_list = List::new(items)
        .block(Block::default().borders(Borders::ALL))
        .highlight_style(highlight_style);

    let mut list_state = ListState::default().with_selected(Some(ac.selected));

    // Clear the popup area first so the overlay has a clean background.
    frame.render_widget(ratatui::widgets::Clear, popup_area);
    frame.render_stateful_widget(popup_list, popup_area, &mut list_state);
}
```
