Route key events to the editor FIRST; intercept Enter and Esc BEFORE passing to the editor:
```rust
AppMode::Adding | AppMode::Editing { .. } => match key.code {
    // Esc cancels — discard editor content, return to Normal (D-05).
    KeyCode::Esc => {
        self.exit_edit_mode();
    }
    // Enter saves — D-12.
    KeyCode::Enter => {
        // Get the first (and only) line from the single-line editor.
        let raw = self.editor.lines().first().cloned().unwrap_or_default();
        let raw = raw.trim().to_string();
        if !raw.is_empty() {
            let task = todotxt_core::Task::parse(&raw);
            match &self.mode.clone() {
                AppMode::Adding => {
                    if let Err(e) = self.task_list.add(task) {
                        eprintln!("add error: {e}");
                    } else {
                        // Select the newly added task (last in list, D-13).
                        self.selected = self.task_list.len().saturating_sub(1);
                    }
                }
                AppMode::Editing { original_idx } => {
                    let idx = *original_idx;
                    if let Err(e) = self.task_list.update(idx, task) {
                        eprintln!("edit error: {e}");
                    }
                    // Keep selected at same idx (D-13).
                }
                _ => unreachable!(),
            }
        }
        self.exit_edit_mode();
    }
    // All other keys — pass to tui-textarea using input_without_shortcuts()
    // to avoid conflicting default bindings (PITFALLS: single-line editors).
    _ => {
        use crossterm::event::Event;
        self.editor.input_without_shortcuts(Event::Key(key));
    }
}
```

```rust
/// Return to Normal mode. Apply any queued reload (D-10, TUI-UX-03).
fn exit_edit_mode(&mut self) {
    self.mode = AppMode::Normal;
    self.clamp_selection();
    if self.pending_reload {
        self.pending_reload = false;
        if let Err(e) = self.task_list.reload() {
            eprintln!("deferred reload error: {e}");
        }
        self.clamp_selection();
    }
}

/// Identical to exit_edit_mode but returns Result (for callers that need ?).
fn apply_pending_reload(&mut self) -> color_eyre::Result<()> {
    if self.pending_reload {
        self.pending_reload = false;
        self.task_list.reload().map_err(|e| {
            color_eyre::eyre::eyre!("deferred reload: {e}")
        })?;
        self.clamp_selection();
    }
    Ok(())
}

/// Clamp `selected` to [0, task_count - 1] (D-07 from Phase 10).
fn clamp_selection(&mut self) {
    let count = self.task_list.len();
    if count > 0 {
        self.selected = self.selected.min(count - 1);
    } else {
        self.selected = 0;
    }
}
```

**Draw method structure:**

```rust
fn draw(&mut self, frame: &mut Frame) {
    use ratatui::layout::{Constraint::{Length, Min}, Layout};
    use ratatui::style::{Modifier, Style};
    use ratatui::text::{Line, Span};
    use ratatui::widgets::{Block, Borders, List, ListItem, ListState, Paragraph};

    match &self.mode {
        AppMode::Normal | AppMode::Adding | AppMode::Editing { .. } => {
            // Two-row layout: task list + footer (D-02, D-14)
            let chunks = Layout::vertical([Min(0), Length(1)]).split(frame.area());
            let list_area = chunks[0];
            let footer_area = chunks[1];

                AppMode::Normal => self.render_status_bar(frame, footer_area),

                    frame.render_widget(&self.editor, footer_area);
```

**Extract task list rendering into a helper:**

```rust
fn render_task_list(&self, frame: &mut Frame, area: ratatui::layout::Rect) {
    use ratatui::style::{Modifier, Style};
    use ratatui::widgets::{List, ListItem, ListState};

    let tasks = self.task_list.tasks();
    let items: Vec<ListItem> = if tasks.is_empty() {
        vec![ListItem::new("(no tasks)")]
    } else {
        tasks.iter().enumerate().map(|(i, t)| {
            let content = format!("{}: {}", i + 1, t.to_raw());
            let style = if t.completed {
                Style::default().add_modifier(Modifier::DIM)
            } else {
                Style::default()
            };
            ListItem::new(content).style(style)
        }).collect()
    };

    let list = List::new(items)
        .highlight_style(Style::default().add_modifier(Modifier::REVERSED));
    let mut list_state = ListState::default();
    if !tasks.is_empty() {
        list_state = list_state.with_selected(Some(self.selected));
    }
    frame.render_stateful_widget(list, area, &mut list_state);
}
```

```rust
            raw.to_string()

        "(no task)".to_string()

        Span::styled("Delete: ", Style::default()),
        Span::raw(format!("\"{}\"", preview)),
        Span::raw("  "),
        Span::styled("y=confirm  any=cancel", Style::default()),

}
```

```rust
fn render_status_bar(&self, frame: &mut Frame, area: ratatui::layout::Rect) {
    use ratatui::text::{Line, Span};
    use ratatui::widgets::Paragraph;
    use todotxt_core::DueStatus;

    let tasks = self.task_list.tasks();
    let total = tasks.len();
    let visible = total;
    let due_today = tasks.iter()
        .filter(|t| !t.completed && t.due_status() == DueStatus::Today)
        .count();
    let overdue = tasks.iter()
        .filter(|t| !t.completed && t.due_status() == DueStatus::Overdue)
        .count();
    let file_name = self.todo_path.file_name()
        .and_then(|n| n.to_str()).unwrap_or("todo.txt");

    let left = format!(
        "{} | {} tasks | {} visible | {} due today | {} overdue",
        file_name, total, visible, due_today, overdue
    );
    // Update key hints to include n/u/d (Phase 11 keys).
    let right = "q quit | n add | u edit | d del | x done | j/k nav";

    let line = Line::from(vec![
        Span::raw(left),
        Span::raw("  "),
        Span::raw(right),
    ]);
    frame.render_widget(Paragraph::new(line), area);
}
```
