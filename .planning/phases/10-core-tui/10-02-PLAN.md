**1. Add toggle_done() method:**
```rust
/// Toggle the completion state of the currently selected task and persist.
/// Per D-10/D-11/D-12: toggles both ways (done→incomplete AND incomplete→done).
/// u is an alias for x (D-12) — both call this method.
fn toggle_done(&mut self) {
    let count = self.task_list.len();
    if count == 0 {
        return;
    }
    let idx = self.selected;
    let task = self.task_list.tasks()[idx].clone();
    let toggled = task.with_completed(!task.completed);
    // update() calls save() internally — single atomic disk write (temp + rename)
    if let Err(e) = self.task_list.update(idx, toggled) {
        // Non-fatal: log to stderr (terminal restore guard is active)
        // Could be extended to show error in status bar in future phase
        eprintln!("toggle_done: {e}");
    }
    // Clamp selected in case reload shrinks the list (e.g., empty file edge case)
    let new_count = self.task_list.len();
    if new_count > 0 {
        self.selected = self.selected.min(new_count - 1);
    } else {
        self.selected = 0;
    }
}
```

```rust
fn draw(&self, frame: &mut Frame) {
    use ratatui::layout::{Constraint::{Length, Min}, Layout};
    use ratatui::style::{Modifier, Style};
    use ratatui::widgets::{List, ListItem, ListState, Paragraph};
    use ratatui::text::{Line, Span};
    use todotxt_core::DueStatus;

    // ── Layout ────────────────────────────────────────────────────────────
    let chunks = Layout::vertical([Min(0), Length(1)]).split(frame.area());
    let list_area   = chunks[0];
    let status_area = chunks[1];

    // ── Task list ─────────────────────────────────────────────────────────
    let tasks = self.task_list.tasks();

    let items: Vec<ListItem> = if tasks.is_empty() {
        vec![ListItem::new("(no tasks)")]
    } else {
        tasks.iter().enumerate().map(|(i, t)| {
            let content = format!("{}: {}", i + 1, t.to_raw());
            let style = if t.completed {
                Style::default().add_modifier(Modifier::DIM)
            } else {
                Style::default()
            };
            ListItem::new(content).style(style)
        }).collect()
    };

    let list = List::new(items)
        .highlight_style(Style::default().add_modifier(Modifier::REVERSED));

    let mut list_state = ListState::default();
    if !tasks.is_empty() {
        list_state = list_state.with_selected(Some(self.selected));
    }

    frame.render_stateful_widget(list, list_area, &mut list_state);

    // ── Status bar (D-14 through D-17) ────────────────────────────────────
    let total   = tasks.len();
    // "visible" in Phase 10 = total (no filtering yet; Phase 11 adds filters)
    let visible = total;
    let due_today = tasks.iter()
        .filter(|t| !t.completed && t.due_status() == DueStatus::Today)
        .count();
    let overdue = tasks.iter()
        .filter(|t| !t.completed && t.due_status() == DueStatus::Overdue)
        .count();

    let file_name = self.todo_path
        .file_name()
        .and_then(|n| n.to_str())
        .unwrap_or("todo.txt");

    let left = format!(
        "{} | {} tasks | {} visible | {} due today | {} overdue",
        file_name, total, visible, due_today, overdue
    );
    let right = "q quit | x done | j/k nav";

    // Single-line status with left and right segments separated by spaces.
    // Agent discretion per D-16: simple Span approach chosen for readability.
    let status_line = Line::from(vec![
        Span::raw(left),
        Span::raw("  "),
        Span::raw(right),
    ]);

    frame.render_widget(Paragraph::new(status_line), status_area);
}
```
