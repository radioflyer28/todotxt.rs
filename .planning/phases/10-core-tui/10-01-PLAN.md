**4. draw() — replace Paragraph with List+ListState (full terminal area for now, Plan 02 adds layout split):**
```rust
fn draw(&self, frame: &mut Frame) {
    use ratatui::widgets::{List, ListItem, ListState};
    use ratatui::style::{Modifier, Style};

    let tasks = self.task_list.tasks();

    let items: Vec<ListItem> = if tasks.is_empty() {
        vec![ListItem::new("(no tasks)")]
    } else {
        tasks.iter().enumerate().map(|(i, t)| {
            let content = format!("{}: {}", i + 1, t.to_raw());
            let style = if t.completed {
                Style::default().add_modifier(Modifier::DIM)
            } else {
                Style::default()
            };
            ListItem::new(content).style(style)
        }).collect()
    };

    let list = List::new(items)
        .highlight_style(Style::default().add_modifier(Modifier::REVERSED));

    let mut state = ListState::default();
    if !tasks.is_empty() {
        state = state.with_selected(Some(self.selected));
    }

    // frame.area() — correct ratatui 0.30 API (not frame.size())
    frame.render_stateful_widget(list, frame.area(), &mut state);
}
```

**5. handle_event() — add navigation branches:**
The existing match already handles Key/Resize/FileChanged/Error. Extend the `AppEvent::Key` branch:

```rust
AppEvent::Key(key) => {
    if key.kind != KeyEventKind::Press {
        return Ok(());
    }
    let task_count = self.task_list.len();
    match key.code {
        // Quit (carried from Phase 9)
        KeyCode::Char('q') => { self.should_quit = true; return Ok(()); }
        KeyCode::Char('c') if key.modifiers.contains(KeyModifiers::CONTROL) => {
            self.should_quit = true; return Ok(());
        }
        // Navigation — only meaningful if list is non-empty
        _ if task_count == 0 => {}
        // Move down 1
        KeyCode::Char('j') | KeyCode::Down => {
            self.selected = (self.selected + 1).min(task_count - 1);
        }
        // Move up 1
        KeyCode::Char('k') | KeyCode::Up => {
            self.selected = self.selected.saturating_sub(1);
        }
        // Jump to first
        KeyCode::Char('g') => { self.selected = 0; }
        // Jump to last
        KeyCode::Char('G') => { self.selected = task_count - 1; }
        // Half-page down (Ctrl+d)
        KeyCode::Char('d') if key.modifiers.contains(KeyModifiers::CONTROL) => {
            let half = (self.list_height / 2) as usize;
            self.selected = (self.selected + half).min(task_count - 1);
        }
        // Half-page up (Ctrl+u)
        KeyCode::Char('u') if key.modifiers.contains(KeyModifiers::CONTROL) => {
            let half = (self.list_height / 2) as usize;
            self.selected = self.selected.saturating_sub(half);
        }
        _ => {}
    }
}
AppEvent::Resize(_, rows) => {
    // Update list_height on resize. Subtract 1 for status bar row (Plan 02).
    self.list_height = rows.saturating_sub(1);
}
```
