/// Paste clipboard content as new task entries (Normal mode, `p` key).
/// Reads clipboard, splits on newlines, filters empty lines, adds each as a new task (D-11, D-12).
/// All lines pasted in a single operation (D-05, D-11).
fn paste_from_clipboard(&mut self) -> color_eyre::Result<()> {
    use crate::app::Task;

    // 1. Lazy-initialize arboard if needed (reuse from Plan 01)
    if self.clipboard.is_none() {
        match Clipboard::new() {
            Ok(cb) => self.clipboard = Some(cb),
            Err(_e) => {
                self.push_runtime_warning("clipboard is empty".to_string()); // D-13
                return Ok(());
            }
        }
    }

    // 2. Read clipboard text; handle empty clipboard (D-13)
    let clipboard_text = if let Some(ref mut cb) = self.clipboard {
        match cb.get_text() {
            Ok(text) => text,
            Err(_e) => {
                self.push_runtime_warning("clipboard is empty".to_string()); // D-13
                return Ok(());
            }
        }
    } else {
        self.push_runtime_warning("clipboard is empty".to_string());
        return Ok(());
    };

    // 3. Split on newlines, filter empty lines (D-11, D-12)
    let lines: Vec<&str> = clipboard_text
        .lines()
        .filter(|line| !line.trim().is_empty())
        .collect();

    if lines.is_empty() {
        self.push_runtime_warning("clipboard is empty".to_string()); // D-13
        return Ok(());
    }

    // 4. Parse each line as a Task and add to task_list (D-12: raw text, no transformation)
    for line in lines {
        match Task::parse(line) {
            Ok(task) => {
                self.task_list.add_task(task)?;
            }
            Err(_e) => {
                // Silently skip invalid lines (user responsible for valid format, D-12)
                // Alternatively, log the error: eprintln!("Invalid task line: {}", line);
            }
        }
    }

    // 5. Rebuild views and reanchor (D-14)
    self.rebuild_all_panes()?;
    self.rebuild_and_reanchor()?;

    // 6. Show feedback
    self.push_runtime_warning(format!("pasted {} task(s)", lines.len()));

    Ok(())
}
```

- D-05, D-11: All lines pasted in one operation
- D-13: Empty clipboard returns silent no-op with hint
- D-14: Rebuild views after task addition
- Graceful error handling: invalid lines skip silently (user responsible for format)

/// Paste clipboard content into the Adding-mode editor (Ctrl+V, D-15).
/// Inserts first clipboard line only (editor is single-line).
/// Empty clipboard is a silent no-op (D-15).
fn paste_in_editor(&mut self) -> color_eyre::Result<()> {
    // 1. Lazy-initialize arboard if needed
    if self.clipboard.is_none() {
        match Clipboard::new() {
            Ok(cb) => self.clipboard = Some(cb),
            Err(_e) => {
                return Ok(()); // Silent no-op on init failure (D-15)
            }
        }
    }

    // 2. Read clipboard text
    let clipboard_text = if let Some(ref mut cb) = self.clipboard {
        match cb.get_text() {
            Ok(text) => text,
            Err(_e) => {
                return Ok(()); // Silent no-op if empty (D-15)
            }
        }
    } else {
        return Ok(());
    };

    // 3. Extract first line only (single-line editor, D-15)
    let first_line = clipboard_text
        .lines()
        .next()
        .unwrap_or("")
        .to_string();

    if !first_line.is_empty() {
        // 4. Insert into editor (D-15)
        self.editor.insert_str(&first_line);
    }

    Ok(())
}
```

**Why:**
- D-15: First line only (editor is single-line in Adding mode)
- D-15: Silent no-op on empty clipboard (no error shown)
- Integrates cleanly with `handle_editor_key` Ctrl+V intercept
- Enables duplicate-and-tweak workflow: `p` in Normal mode pastes entire task, then user presses `n` to edit and presses Ctrl+V to pre-fill (CLIP-04 complete UX)
