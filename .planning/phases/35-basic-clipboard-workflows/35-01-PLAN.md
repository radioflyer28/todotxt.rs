```rust
/// Copy selected or active task text to system clipboard.
/// Targets selected_tasks (if non-empty) or the active cursor task (per D-03).
/// Multi-task copy joins lines with newlines in descending-canonical-index order (per D-08, D-17).
fn copy_selected_to_clipboard(&mut self) -> color_eyre::Result<()> {
    // 1. Determine targets: selected tasks or active task (D-03)
    let mut targets: Vec<usize> = if !self.selected_tasks.is_empty() {
        self.selected_tasks.iter().copied().collect()
    } else {
        if let Some(DisplayRow::Task(idx)) = self.display_rows.get(self.selected) {
            vec![*idx]
        } else {
            vec![]
        }
    };

    // Abort if no task is selected (D-10: skip header rows silently)
    if targets.is_empty() {
        self.push_runtime_warning("No task selected".to_string());
        return Ok(());
    }

    // 2. Sort targets in descending-canonical-index order (D-08, D-17)
    targets.sort_unstable_by(|a, b| b.cmp(a));

    // 3. Collect raw text from each target task
    let tasks = self.task_list.tasks();
    let mut clipboard_text = Vec::new();
    for &idx in &targets {
        if let Some(task) = tasks.get(idx) {
            clipboard_text.push(task.to_raw());
        }
    }
    let text_to_copy = clipboard_text.join("\n");

    // 4. Lazy-initialize arboard if needed (D-02)
    if self.clipboard.is_none() {
        match Clipboard::new() {
            Ok(cb) => self.clipboard = Some(cb),
            Err(_e) => {
                self.push_runtime_warning("Clipboard unavailable".to_string());
                return Ok(()); // Silent graceful no-op (D-02)
            }
        }
    }

    // 5. Set clipboard content; show status feedback (D-09)
    if let Some(ref mut cb) = self.clipboard {
        match cb.set_text(text_to_copy) {
            Ok(_) => {
                let count_msg = if targets.len() == 1 {
                    "copied 1 task".to_string()
                } else {
                    format!("copied {} tasks", targets.len())
                };
                self.push_runtime_warning(count_msg);
            }
            Err(_e) => {
                self.push_runtime_warning("Failed to copy to clipboard".to_string());
            }
        }
    }

    Ok(())
}
```

**Why:**
- D-03: Targets selected_tasks or active cursor (same semantics as `s` and `i`)
- D-08, D-17: Multi-task copy joins with newlines in descending index order
- D-09: Status feedback via push_runtime_warning
- D-02: Lazy initialization of arboard, graceful error handling
- D-10: Only DisplayRow::Task entries contribute (empty targets are skipped)
