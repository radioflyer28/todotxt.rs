```rust
fn handle_filter_defining_key(&mut self, key: KeyEvent) {
    use crossterm::event::KeyCode;
    let state = match self.filter_defining_state.as_mut() {
        Some(s) => s,
        None => return,
    };

    match key.code {
        // D-03: Esc = discard, nothing written to TOML
        KeyCode::Esc => {
            self.filter_defining_state = None;
            self.mode = AppMode::Normal;
        }

        // D-04: Enter = save preset definitions to TOML, then return to Normal
        KeyCode::Enter => {
            // Flush active editor content back to filter_query (live preview finalised)
            self.filter_query = state.active_editor.lines().join("").trim().to_string();

            // Update config.presets from editors
            for (i, name) in state.preset_names.iter().enumerate() {
                let filter_str = state.preset_editors[i].lines().join("").trim().to_string();
                self.config.presets.entry(name.clone())
                    .and_modify(|p| p.filter = if filter_str.is_empty() { None } else { Some(filter_str.clone()) })
                    .or_insert_with(|| crate::config::TuiPreset { filter: if filter_str.is_empty() { None } else { Some(filter_str) } });
            }

            // Rebuild presets vec from updated config (D-05: only preset strings persisted)
            let mut updated: Vec<(String, String)> = self.config.presets.iter()
                .map(|(k, v)| (k.clone(), v.filter.clone().unwrap_or_default()))
                .collect();
            updated.sort_by_key(|(name, _)| name.clone());
            self.presets = updated;

            // Persist to TOML
            if let Some(ref path) = self.config_path {
                if let Err(e) = self.config.save(path) {
                    eprintln!("Warning: failed to save config: {e}");
                }
            }

            self.filter_defining_state = None;
            self.mode = AppMode::Normal;
            self.rebuild_display_indices();
        }

        // Navigate rows (Up/Down)
        KeyCode::Up => {
            if state.selected_row > 0 {
                state.selected_row -= 1;
            }
        }
        KeyCode::Down => {
            let max_row = state.preset_editors.len(); // 0 = active row, 1..=N = presets
            if state.selected_row < max_row {
                state.selected_row += 1;
            }
        }

        // All other keys: forward to the focused editor
        _ => {
            if state.selected_row == 0 {
                state.active_editor.input(key);
                // D-07: live preview — update filter_query from active editor for rebuild
                self.filter_query = state.active_editor.lines().join("").trim().to_string();
                self.rebuild_display_indices();
            } else {
                let idx = state.selected_row - 1;
                if idx < state.preset_editors.len() {
                    state.preset_editors[idx].input(key);
                }
            }
        }
    }
}
```

```rust
fn render_filter_defining_panel(&mut self, frame: &mut Frame, area: Rect) {
    use ratatui::widgets::{Block, Borders, List, ListItem};
    use ratatui::style::{Modifier, Style};

    let state = match self.filter_defining_state.as_mut() {
        Some(s) => s,
        None => return,
    };

    let outer = Block::default()
        .title(" Filter Definitions (F) — Enter: save · Esc: discard ")
        .borders(Borders::ALL);
    let inner = outer.inner(area);
    frame.render_widget(outer, area);

    // Split inner area: row 0 = active filter editor, remaining = preset list
    let row_height = 1u16;
    let rows = Layout::default()
        .direction(Direction::Vertical)
        .constraints([
            Constraint::Length(row_height + 1),  // active filter row + separator
            Constraint::Min(1),                  // preset list
        ])
        .split(inner);

    // --- Row 0: Active filter (editable, D-07) ---
    let active_focused = state.selected_row == 0;
    let active_block = Block::default()
        .title(" Active filter ")
        .borders(Borders::BOTTOM)
        .border_style(if active_focused {
            Style::default().add_modifier(Modifier::BOLD)
        } else {
            Style::default()
        });
    // Render the active_editor TextArea widget
    state.active_editor.set_block(active_block);
    frame.render_widget(state.active_editor.widget(), rows[0]);

    // --- Rows 1–9: Preset list ---
    let items: Vec<ListItem> = state.preset_names.iter().enumerate().map(|(i, name)| {
        let filter_val = state.preset_editors[i].lines().join("");
        let label = format!(" #{} {}  {}", i + 1, name, filter_val);
        let style = if state.selected_row == i + 1 {
            Style::default().add_modifier(Modifier::REVERSED)
        } else {
            Style::default()
        };
        ListItem::new(label).style(style)
    }).collect();

    let preset_list = List::new(items)
        .block(Block::default().title(" Presets ").borders(Borders::NONE));
    frame.render_widget(preset_list, rows[1]);
}
```
