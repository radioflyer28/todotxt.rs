```rust
/// Returns true if `filter` is a single @/+ tag token with no spaces.
/// Valid: "@work", "+project", "@home-office". Invalid: "@work @home", "due:today", "".
/// Used to validate pane filters before pane task movement (Phase 41, PMOVE-01, D-15).
fn is_single_tag_token(filter: &str) -> bool {
    if filter.is_empty() {
        return false;
    }
    let trimmed = filter.trim();
    // Must start with @ or + and contain no whitespace.
    (trimmed.starts_with('@') || trimmed.starts_with('+'))
        && !trimmed.contains(char::is_whitespace)
}
```

2. Add the `pane_move_task` method on App:

```rust
/// Move task(s) from the active pane to an adjacent pane by tag mutation (Phase 41, PMOVE-02).
///
/// `direction` is +1 (right) or -1 (left). Wraps at boundaries.
///
/// Validation (PMOVE-03, D-15):
/// - Source pane must have a single-token filter (@/+ tag).
/// - Dest pane must have a single-token filter (@/+ tag).
/// - If either validation fails, push a status bar message and return Ok(()).
///
/// Mutation per task (D-16):
/// - Remove the source pane's filter token from the task's raw text (if present).
/// - Append the dest pane's filter token (if not already present).
///
/// After mutation (D-17, D-18):
/// - Push undo entry BEFORE mutation.
/// - Save task list.
/// - Rebuild all panes.
/// - Jump active_pane to dest pane index.
fn pane_move_task(&mut self, direction: isize) -> color_eyre::Result<()> {
    let pane_count = self.panes.len();
    if pane_count < 2 {
        self.set_status_message("Need at least 2 panes to move tasks.".to_string());
        return Ok(());
    }

    let src_idx = self.active_pane;
    let dest_idx = ((src_idx as isize + direction).rem_euclid(pane_count as isize)) as usize;

    let src_filter = self.panes[src_idx].filter_query.trim().to_string();
    let dest_filter = self.panes[dest_idx].filter_query.trim().to_string();

    // Validate source filter (PMOVE-03, D-15).
    if !is_single_tag_token(&src_filter) {
        self.set_status_message(format!(
            "Cannot move: source pane filter '{}' is not a single @/+ tag.",
            src_filter
        ));
        return Ok(());
    }
    // Validate dest filter (PMOVE-03, D-15).
    if !is_single_tag_token(&dest_filter) {
        self.set_status_message(format!(
            "Cannot move: destination pane filter '{}' is not a single @/+ tag.",
            dest_filter
        ));
        return Ok(());
    }

    // Collect global task indices: selected_tasks if non-empty, else cursor task.
    let display_indices = &self.panes[src_idx].display_indices;
    let cursor = self.panes[src_idx].cursor;

    let task_indices: Vec<usize> = if !self.selected_tasks.is_empty() {
        self.selected_tasks.iter().cloned().collect()
    } else if cursor < display_indices.len() {
        vec![display_indices[cursor]]
    } else {
        self.set_status_message("No task selected to move.".to_string());
        return Ok(());
    };

    if task_indices.is_empty() {
        return Ok(());
    }

    // Push undo before mutation (D-17).
    self.push_undo_entry();

    // Mutate each task: remove src_filter token, append dest_filter token.
    for &task_idx in &task_indices {
        if let Some(task) = self.task_list.tasks.get_mut(task_idx) {
            let mut raw = task.raw_text.clone();

            // Remove source filter token (word-boundary safe: split by whitespace).
            let tokens: Vec<&str> = raw.split_whitespace().collect();
            let filtered: Vec<&str> = tokens
                .iter()
                .copied()
                .filter(|&t| !t.eq_ignore_ascii_case(&src_filter))
                .collect();
            raw = filtered.join(" ");

            // Append dest filter token if not already present.
            let already_has_dest = raw
                .split_whitespace()
                .any(|t| t.eq_ignore_ascii_case(&dest_filter));
            if !already_has_dest {
                if !raw.is_empty() {
                    raw.push(' ');
                }
                raw.push_str(&dest_filter);
            }

            task.raw_text = raw;
        }
    }

    // Persist and rebuild (D-18).
    self.save_task_list()?;
    self.selected_tasks.clear();
    self.selection_anchor = None;
    self.active_pane = dest_idx;
    self.rebuild_all_panes();
    self.rebuild_display_indices();

    Ok(())
}
```

```rust
fn pane_move_task_tag_swap() {
    use std::collections::HashMap;
    use crate::config::{TuiConfig, PaneConfig, PaneSort};
    let mut config = TuiConfig::default();
    config.panes = vec![
```
