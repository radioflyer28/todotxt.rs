Output: Pane-scoped filter routing, per-pane filter application in task display, filter state preservation across pane switches.

pub fn rebuild_visible_rows(&mut self) {

        0

        self.active_pane
    };
    let pane = &mut self.panes[pane_idx];

    // Per-pane query behavior (D-04, Phase 25): Apply active pane's filter_query
    let filter_str = &pane.filter_query.trim();

    // Apply filter to task list, getting (source_index, task) tuples
    let mut rows: Vec<DisplayRow> = self
        .task_list
        .filter(&filter)
        .into_iter()
        .map(|(source_index, _task)| DisplayRow::Task(source_index))
        .collect();

    // TODO Phase 25-02: Apply pane.sort_order and pane.grouping here
    // For now, rows remain in filtered order (no sort/group applied)

    pane.display_rows = rows;

    // Reconcile selection: clamp to bounds of new display_rows
    if pane.selected >= pane.display_rows.len() && !pane.display_rows.is_empty() {

    } else if pane.display_rows.is_empty() {
        pane.selected = 0;
    }
}
