### 5. Add handle_filtering_key() method

```rust
fn handle_filtering_key(
    &mut self,
    key: crossterm::event::KeyEvent,
) -> color_eyre::Result<()> {
    match key.code {
        KeyCode::Esc => {
            // D-05: Esc closes AND clears the filter.
            self.filter_query = String::new();
            self.filter_state = None;
            self.mode = AppMode::Normal;
            self.rebuild_and_reanchor();
            self.apply_pending_reload()?;
        }
        KeyCode::Enter => {
            // Enter closes the panel but keeps the active filter (D-05 exception).
            self.filter_state = None;
            self.mode = AppMode::Normal;
            self.apply_pending_reload()?;
        }
        KeyCode::Down => {
            if let Some(ref mut state) = self.filter_state {
                if !self.presets.is_empty() {
                    state.selected_preset =
                        (state.selected_preset + 1).min(self.presets.len() - 1);
                    let query = self.presets[state.selected_preset].1.clone();
                    // Sync query into editor and filter_query.
                    state.editor = TextArea::default();
                    state.editor.insert_str(&query);
                    self.filter_query = query;
                    self.rebuild_and_reanchor();
                }
            }
        }
        KeyCode::Up => {
            if let Some(ref mut state) = self.filter_state {
                if !self.presets.is_empty() && state.selected_preset > 0 {
                    state.selected_preset = state.selected_preset.saturating_sub(1);
                    let query = self.presets[state.selected_preset].1.clone();
                    state.editor = TextArea::default();
                    state.editor.insert_str(&query);
                    self.filter_query = query;
                    self.rebuild_and_reanchor();
                }
            }
        }
        KeyCode::Char(c @ '1'..='9') => {
            let idx = (c as usize) - ('1' as usize);
            if idx < self.presets.len() {
                let query = self.presets[idx].1.clone();
                if let Some(ref mut state) = self.filter_state {
                    state.editor = TextArea::default();
                    state.editor.insert_str(&query);
                    state.selected_preset = idx;
                }
                self.filter_query = query;
                self.rebuild_and_reanchor();
            }
        }
        _ => {
            // Pass all other keys to the tui-textarea filter editor.
            if let Some(ref mut state) = self.filter_state {
                state.editor.input_without_shortcuts(Event::Key(key));
                // Sync editor content to filter_query.
                self.filter_query = state.editor
                    .lines()
                    .first()
                    .cloned()
                    .unwrap_or_default();
            }
            self.rebuild_and_reanchor();
        }
    }
    Ok(())
}
```

BORROW NOTE: `self.presets` and `self.filter_state` are both on self. When you need both simultaneously (Down/Up arms), extract the preset index and query from filter_state first as owned Strings, then update self fields. The pattern above uses `.clone()` on the query string to avoid borrow conflicts.

### 6. Add render_filter_panel() method

```rust
fn render_filter_panel(&mut self, frame: &mut Frame, area: ratatui::layout::Rect) {
    use ratatui::layout::Rect;
    use ratatui::style::{Modifier, Style};
    use ratatui::widgets::{List, ListItem, ListState};

    // Row 0: tui-textarea text input (border-less, 1 row).
    let input_area = Rect { height: 1, ..area };
    if let Some(ref state) = self.filter_state {
        frame.render_widget(&state.editor, input_area);
    }

    // Rows 1..: numbered preset list (if any presets configured).
    if area.height > 1 && !self.presets.is_empty() {
        let list_area = Rect {
            y: area.y + 1,
            height: area.height - 1,
            ..area
        };
        let selected_preset = self.filter_state.as_ref().map(|s| s.selected_preset);
        let items: Vec<ListItem> = self
            .presets
            .iter()
            .enumerate()
            .map(|(i, (name, query))| {
                ListItem::new(format!("{}. {} — {}", i + 1, name, query))
            })
            .collect();
        let list = List::new(items)
            .highlight_style(Style::default().add_modifier(Modifier::REVERSED));
        let mut list_state = ListState::default().with_selected(selected_preset);
        frame.render_stateful_widget(list, list_area, &mut list_state);
    }
}
```
