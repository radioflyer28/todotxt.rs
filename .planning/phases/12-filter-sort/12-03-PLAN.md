```rust
fn render_status_bar(&self, frame: &mut Frame, area: ratatui::layout::Rect) {
    use ratatui::text::{Line, Span};
    use ratatui::widgets::Paragraph;
    use todotxt_core::DueStatus;

    let tasks = self.task_list.tasks();
    let total = tasks.len();
    let visible = self.display_indices.len();

    // Due-today and overdue counts computed from visible tasks only (display_indices).
    let due_today = self
        .display_indices
        .iter()
        .filter(|&&ci| {
            !tasks[ci].completed && tasks[ci].due_status() == DueStatus::Today
        })
        .count();
    let overdue = self
        .display_indices
        .iter()
        .filter(|&&ci| {
            !tasks[ci].completed && tasks[ci].due_status() == DueStatus::Overdue
        })
        .count();

    let file_name = self
        .todo_path
        .file_name()
        .and_then(|n| n.to_str())
        .unwrap_or("todo.txt");

    // Build left section: always shown.
    let mut left = format!("{} | {}/{} tasks", file_name, visible, total);
    if due_today > 0 || overdue > 0 {
        left.push_str(&format!(" | {} due today | {} overdue", due_today, overdue));
    }

    // Build middle section: shown only when filter or sort is active.
    let mut middle = String::new();
    if !self.filter_query.trim().is_empty() {
        middle.push_str(&format!(" | {}", self.filter_query.trim()));
    }
    if self.sort_order != SortOrder::FileOrder {
        middle.push_str(&format!(" | sort: {}", sort_name(self.sort_order)));
    }

    // Build right section: key hints (lowest priority for truncation).
    let right = "  q quit | n add | u edit | d del | x done | j/k nav | f filter | o sort";

    // Truncate middle (filter query) with "…" if combined line exceeds terminal width.
    let total_width = area.width as usize;
    let left_len = left.len();
    let right_len = right.len();
    let available_middle = total_width.saturating_sub(left_len + right_len);

    let middle_display = if middle.len() > available_middle && available_middle > 1 {
        format!("{}…", &middle[..available_middle.saturating_sub(1)])
    } else {
        middle.clone()
    };

    // If even left + right_len doesn't fit, omit hints entirely.
    let status_line = if left_len + middle_display.len() + right_len <= total_width {
        Line::from(vec![
            Span::raw(left),
            Span::raw(middle_display),
            Span::raw(right),
        ])
    } else {
        Line::from(vec![
            Span::raw(left),
            Span::raw(middle_display),
        ])
    };

    frame.render_widget(Paragraph::new(status_line), area);
}
```
