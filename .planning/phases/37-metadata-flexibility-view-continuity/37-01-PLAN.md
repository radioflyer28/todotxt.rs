**Step 2: Update Filter::from_query parser (lines 55-83)**

Replace the token parsing logic to detect prefix patterns. Parsing rules:
- Token `@foo` (no slash, starts with @) → `ContextPrefix("foo")`
- Token `@email/waiting` (slash present) → Keep as `Include("@email/waiting")` for exact match
- Token `-@foo` (negated, no slash) → `NegContextPrefix("foo")`
- Token `-@email/waiting` (negated with slash) → Keep as `Exclude("@email/waiting")`
- Token `+client` (no slash) → `ProjectPrefix("client")`
- Token `+client/acme` (slash present) → `Include("+client/acme")`
- Token `-+client` → `NegProjectPrefix("client")`
- Token `-+client/acme` → `Exclude("-+client/acme")`

Insert this logic before the current negation/Include fallback:
```rust
// Check for context prefix tokens (@context or -@context)
if let Some(rest) = token.strip_prefix('@') {
    // Check if negated
    if let Some(context_name) = token.strip_prefix("-@") {
        // Negated context prefix
        if !context_name.contains('/') {
            // @foo with no slash → NegContextPrefix
            return FilterTerm::NegContextPrefix(context_name.to_string());
        }
    } else if !rest.contains('/') {
        // @foo with no slash → ContextPrefix
        return FilterTerm::ContextPrefix(rest.to_string());
    }
}

// Check for project prefix tokens (+project or -+project)
if let Some(rest) = token.strip_prefix('+') {
    // Check if negated
    if let Some(project_name) = token.strip_prefix("-+") {
        // Negated project prefix
        if !project_name.contains('/') {
            // +foo with no slash → NegProjectPrefix
            return FilterTerm::NegProjectPrefix(project_name.to_string());
        }
    } else if !rest.contains('/') {
        // +foo with no slash → ProjectPrefix
        return FilterTerm::ProjectPrefix(rest.to_string());
    }
}
```

**Step 3: Implement matching logic in matches_with_date (lines 94-120)**

In the match statement on term (around line 108), add four new arms:

```rust
FilterTerm::ContextPrefix(prefix) => {
    let prefix_lower = prefix.to_ascii_lowercase();
    task.contexts.iter().any(|ctx| {
        let ctx_lower = ctx.to_ascii_lowercase();
        ctx_lower == prefix_lower || ctx_lower.starts_with(&format!("{}/", prefix_lower))
    })
}
FilterTerm::ProjectPrefix(prefix) => {
    let prefix_lower = prefix.to_ascii_lowercase();
    task.projects.iter().any(|proj| {
        let proj_lower = proj.to_ascii_lowercase();
        proj_lower == prefix_lower || proj_lower.starts_with(&format!("{}/", prefix_lower))
    })
}
FilterTerm::NegContextPrefix(prefix) => {
    let prefix_lower = prefix.to_ascii_lowercase();
    !task.contexts.iter().any(|ctx| {
        let ctx_lower = ctx.to_ascii_lowercase();
        ctx_lower == prefix_lower || ctx_lower.starts_with(&format!("{}/", prefix_lower))
    })
}
FilterTerm::NegProjectPrefix(prefix) => {
    let prefix_lower = prefix.to_ascii_lowercase();
    !task.projects.iter().any(|proj| {
        let proj_lower = proj.to_ascii_lowercase();
        proj_lower == prefix_lower || proj_lower.starts_with(&format!("{}/", prefix_lower))
    })
}
```
