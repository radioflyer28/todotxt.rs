```rust
mod view_continuity {
    use todotxt_tui::app::App;
    use todotxt_core::task_list::TaskList;
    use std::path::PathBuf;

    // Helper to create a test app with a single pane and initial state
    fn setup_app_with_state(todos: &[&str], filter_query: &str, sort_order: &str, grouping: bool) -> App {
        let mut app = App::new_test();
        // Load initial tasks into TaskList
        let task_list = TaskList::from_strings(todos.to_vec());
        app.task_list = task_list;
        
        // Set up the pane with filter/sort/group state
        if let Some(pane) = app.panes.get_mut(0) {
            pane.filter_query = filter_query.to_string();
            pane.sort_order = sort_order.to_string(); // e.g., "priority" or "due_date"
            pane.grouping = grouping;
        }
        app
    }

    // Helper to assert pane state is preserved
    fn assert_state_preserved(app: &App, expected_filter: &str, expected_sort: &str, expected_grouping: bool, context: &str) {
        if let Some(pane) = app.panes.get(0) {
            assert_eq!(pane.filter_query, expected_filter, "{}: filter_query mismatch", context);
            assert_eq!(pane.sort_order, expected_sort, "{}: sort_order mismatch", context);
            assert_eq!(pane.grouping, expected_grouping, "{}: grouping mismatch", context);
        } else {
            panic!("No pane found in app");
        }
    }
}
```

```rust
#[test]
fn undo_restores_original_raw_text_including_tag_order() {
    let mut app = App::new_test();
    
    // Load a task with tags in specific order
    let task_list = TaskList::from_strings(vec![
        "Buy milk @home +shopping @evening due:2026-05-15".to_string(),
    ]);
    app.task_list = task_list;
    
    // Get original raw text
    let original_raw = app.task_list.tasks()[0].to_raw();
    
    // Edit the task: add priority (which will use rebuild_raw and move tags to end)
    app.set_selected_index(0);
    app.input_buffer.clear();
    app.input_buffer.push_str("(A) Buy milk"); // Only rebuild_raw will add tags back, potentially reordered
    app.save_and_exit_editing_mode();
    
    // Verify the raw text may have tags reordered (rebuild_raw moves them to end)
    let after_edit_raw = app.task_list.tasks()[0].to_raw();
    
    // Undo
    app.apply_undo();
    
    // Verify original raw is restored exactly (including original tag order)
    let restored_raw = app.task_list.tasks()[0].to_raw();
    assert_eq!(restored_raw, original_raw, 
        "Undo must restore the original raw text exactly.\n\
         Original: {}\n\
         After edit: {}\n\
         After undo: {}",
        original_raw, after_edit_raw, restored_raw);
}
```

This test validates D-08 behavior: `UndoEntry.tasks` captures the full `Task` slice (including verbatim `raw` field) before mutation. After `apply_undo()` + `replace_all()`, the original raw is written back to disk.
