/// Prepend `text` before the task body, after completion marker, priority, and date prefixes.
///
/// The inserted text is separated from the existing body by a single space.
/// Tags (`+project`, `@context`, `due:`, `t:`) in the prepended text are parsed normally.
///
/// # Example
/// ```
/// let task = Task::parse("Call dentist @personal");
/// let updated = task.with_text_prepended("(A)");
/// assert!(updated.to_raw().starts_with("(A) Call dentist"));
/// ```
pub fn with_text_prepended(self, text: &str) -> Self {
    let new_body = format!("{} {}", text, self.body);
    let new_task = Task { body: new_body, ..self };
    let new_raw = rebuild_raw(&new_task);
    Task::parse(&new_raw)
}
```

Then add two unit tests to `crates/todotxt-core/tests/task_tests.rs`:

```rust
#[test]
fn with_text_prepended_inserts_before_body() {
    let task = Task::parse("Call dentist @personal");
    let updated = task.with_text_prepended("(A)");
    // Priority parsed from prepended text, body starts with original body
    assert!(updated.to_raw().starts_with("(A) Call dentist"));
    assert_eq!(updated.priority, Some('A'));
}

#[test]
fn with_text_prepended_preserves_completion_and_dates() {
    // For a plain incomplete task with creation date, prepend should not disturb the date
    let task = Task::parse("2026-01-01 Buy milk +groceries");
    let updated = task.with_text_prepended("NOTE:");
    // raw: "2026-01-01 NOTE: Buy milk +groceries" — creation date still present
    assert!(updated.to_raw().contains("2026-01-01"));
    assert!(updated.to_raw().contains("NOTE:"));
    assert!(updated.to_raw().contains("Buy milk"));
}
```

/// Print the result of a write operation (add, edit, append, prepend, complete, undo, del).
///
/// Human mode: print `info` to stderr (unless `--quiet`), then print task raw line to stdout.
/// JSON mode: print task in standard envelope; no info line.
pub fn print_write_result(&self, info: &str, idx: usize, task: &Task) {
    if self.json {
        println!("{}", json_success(task_dto(idx, task)));
    } else {
        if !self.quiet {
            eprintln!("{}", info);
        }
        println!("{}", task.to_raw());
    }
}
