#[test]
fn add_creates_task_in_file() {

        .assert()
        .success()

#[test]
fn add_prints_new_task_id_to_stderr() {
    let fixture = TestFixture::new();
    // SAMPLE_TODO has 4 tasks; new task should be ID 5
    fixture
        .cmd()
        .args(["add", "New task"])
        .assert()
        .success()
        .stderr(contains("Added task #5."));
}

#[test]
fn add_with_date_flag_prepends_creation_date() {
    let fixture = TestFixture::new();
    let output = fixture
        .cmd()
        .args(["add", "--date", "Dated task"])
        .assert()
        .success()
        .get_output()
        .stdout
        .clone();
    let stdout = String::from_utf8_lossy(&output);
    // stdout should contain a date pattern YYYY-MM-DD followed by the task text
    assert!(
        stdout.contains("Dated task"),
        "task text must appear in output"
    );
    // Verify date was prepended in the file
    let content = fs::read_to_string(fixture.todo.path()).unwrap();
    let last_line = content.lines().last().unwrap_or("");
    assert!(
        last_line.starts_with("20") && last_line.contains('-'),
        "last line should start with YYYY-MM-DD date: {last_line}"
    );
}

#[test]
fn add_with_no_date_flag_suppresses_date() {
    // Create a fixture with auto_creation_date = true in config
    let fixture = TestFixture::new();
    // Use --no-date flag — even if config had date enabled, this should suppress it
    fixture
        .cmd()
        .args(["add", "--no-date", "No date task"])
        .assert()
        .success();
    let content = fs::read_to_string(fixture.todo.path()).unwrap();
    let last_line = content.lines().last().unwrap_or("");
    // Line should NOT start with a date pattern
    assert!(
        !last_line.starts_with("20") || !last_line.contains('-'),
        "line should NOT have date prefix when --no-date used: {last_line}"
    );
}
