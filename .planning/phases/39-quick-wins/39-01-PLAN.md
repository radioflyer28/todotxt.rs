   ```rust
   fn archive_tasks(&mut self) -> color_eyre::Result<usize> {
       use std::io::Write;
       let done_path = self.archive_path();

       let completed: Vec<_> = self.task_list.tasks().iter()
           .filter(|t| t.completed)
           .cloned()
           .collect();
       let count = completed.len();
       if count == 0 {
           return Ok(0);
       }

       // Ensure done.txt parent exists.
       if let Some(parent) = done_path.parent() {
           std::fs::create_dir_all(parent)?;
       }

       // Read existing done.txt content (or empty string if absent).
       let existing = if done_path.exists() {
           std::fs::read_to_string(&done_path)?
       } else {
           String::new()
       };

       // Build new done.txt: existing + newly completed tasks.
       let appended = completed.iter().map(|t| t.to_raw()).collect::<Vec<_>>().join("\n");
       let new_done = if existing.is_empty() {
           format!("{appended}\n")
       } else {
           let base = existing.trim_end_matches('\n');
           format!("{base}\n{appended}\n")
       };

       // Write-first: atomic write to done.txt before touching todo.txt.
       let done_parent = done_path.parent().unwrap_or(std::path::Path::new("."));
       let mut temp_done = tempfile::NamedTempFile::new_in(done_parent)?;
       temp_done.write_all(new_done.as_bytes())?;
       temp_done.flush()?;
       temp_done.as_file().sync_all()?;
       temp_done.persist(&done_path)
           .map_err(|e| color_eyre::eyre::eyre!("Failed to write done.txt: {}", e.error))?;

       // Only after done.txt succeeds: push undo snapshot and remove completed from task_list.
       self.push_undo_entry();
       let incomplete_indices: Vec<usize> = self.task_list.tasks().iter()
           .enumerate()
           .filter(|(_, t)| !t.completed)
           .map(|(i, _)| i)
           .collect();
       // Rebuild task_list with only incomplete tasks (descending to avoid index shift).
       let mut sorted_desc: Vec<usize> = self.task_list.tasks().iter()
           .enumerate()
           .filter(|(_, t)| t.completed)
           .map(|(i, _)| i)
           .collect();
       sorted_desc.sort_unstable_by(|a, b| b.cmp(a));
       for idx in sorted_desc {
           self.task_list.delete(idx)
               .map_err(|e| color_eyre::eyre::eyre!("Failed to delete archived task {}: {}", idx, e))?;
       }

       self.rebuild_all_panes();
       self.rebuild_and_reanchor();
       self.selected_tasks.clear();
       Ok(count)
   }
   ```
