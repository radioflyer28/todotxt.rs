   ```rust
   fn bulk_mark_done(&mut self) {
       self.push_undo_entry();
       let indices: Vec<usize> = self.selected_tasks.iter().copied().collect();
       let mut marked = 0usize;
       for idx in &indices {
           if let Some(task) = self.task_list.tasks().get(*idx) {
               if !task.completed {
                   let mut updated = task.clone();
                   // Use the same toggle mechanism as toggle_done() in this file.
                   // If toggle_done uses `task.toggle()` call that here;
                   // if it uses task_list.complete(idx), call that here instead.
                   // Check pane_toggle_done to confirm exact API.
                   updated.completed = true;
                   // Add today's completion date if Task has a completion_date field,
                   // matching what toggle_done does. Use chrono::Local::now().date_naive()
                   // only if toggle_done sets it.
                   if let Err(e) = self.task_list.update(*idx, updated) {
                       eprintln!("bulk_mark_done error on idx {idx}: {e}");
                   } else {
                       marked += 1;
                   }
               }
           }
       }
       self.selected_tasks.clear();
       self.runtime_warnings.push(format!("Marked {} task(s) done", marked));
       self.rebuild_all_panes();
       self.rebuild_and_reanchor();
   }
   ```
