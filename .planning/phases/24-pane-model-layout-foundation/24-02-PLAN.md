//! Pane list widget — renders a single pane with its task list

use ratatui::{
    backend::Backend,
    layout::Rect,
    style::{Color, Modifier, Style},
    text::Line,
    widgets::{Block, Borders, List, ListItem, Paragraph},
    Frame,
};
use crate::state::Pane;
use crate::app::DisplayRow;
use crate::theme::StyleSheet;

pub struct PaneList;

impl PaneList {
    /// Render a single pane into the given area
    pub fn render(
        frame: &mut Frame,
        area: Rect,
        pane: &Pane,
        is_active: bool,
        stylesheet: &StyleSheet,
        mode_allows_nav: bool,
    ) {
        // Border style depends on active state
        let border_style = if is_active {
            Style::default()
                .fg(stylesheet.theme.active_pane_border)
                .add_modifier(Modifier::BOLD)
        } else {
            Style::default()
                .fg(stylesheet.theme.inactive_pane_border)
        };

        let title = if is_active {
            format!("▶ {}", pane.label)  // Visual indicator for active pane
        } else {
            format!("  {}", pane.label)
        };

        let block = Block::default()
            .title(title)
            .borders(Borders::ALL)
            .border_style(border_style);

        // Build list items from display_rows
        let items: Vec<ListItem> = pane
            .display_rows
            .iter()
            .enumerate()
            .map(|(idx, row)| {
                let text = match row {
                    DisplayRow::Task(source_idx) => {
                        format!("[{}]", source_idx + 1)  // Show task ID
                    }
                    DisplayRow::GroupHeader(label) => {
                        format!("─ {} ─", label)  // Group headers
                    }
                };

                let item_style = if idx == pane.selected && mode_allows_nav {
                    Style::default()
                        .bg(stylesheet.theme.selected_bg)
                        .fg(stylesheet.theme.selected_fg)
                        .add_modifier(Modifier::BOLD)
                } else {
                    Style::default()
                };

                ListItem::new(Line::from(text).style(item_style))
            })
            .collect();

        let list = List::new(items)
            .block(block)
            .style(Style::default().fg(Color::White));

        frame.render_widget(list, area);
    }
}
```

// OLD (single pane):
// let list_area = layout_result.get_task_list_area();
// task_list_widget.render(frame, list_area, ...);

// NEW (multiple panes):
use ratatui::layout::{Constraint, Direction};

let panes = &app.panes;
if !panes.is_empty() {
    // Calculate width for each pane
    let pane_count = panes.len();
    let list_area = layout_result.get_task_list_area();  // Full task list area
    
    // Divide the list area vertically
    let pane_constraints = vec![Constraint::Percentage(100 / pane_count as u16); pane_count];
    let pane_areas = Layout::default()
        .direction(Direction::Horizontal)
        .constraints(pane_constraints)
        .split(list_area);
    
    // Render each pane
    for (pane_idx, pane) in panes.iter().enumerate() {
        let pane_area = pane_areas[pane_idx];
        let is_active = pane_idx == app.active_pane;
        
        PaneList::render(
            frame,
            pane_area,
            pane,
            is_active,
            &app.theme,
            matches!(app.mode, AppMode::Normal),
        );
    }
}
```
