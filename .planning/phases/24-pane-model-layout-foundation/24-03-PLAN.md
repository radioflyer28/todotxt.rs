impl App {
    /// Determine if we should render in single-pane fallback mode
    pub fn should_show_single_pane(&self) -> bool {
        // Conditions that trigger single-pane mode:
        // 1. No panes defined
        if self.panes.is_empty() {
            return true;
        }
        
        // 2. Exactly one pane (no need for multi-pane UI)
        if self.panes.len() == 1 {
            return true;
        }
        
        // 3. All panes are empty (no tasks to show, confusing multi-pane layout)
        if self.panes.iter().all(|p| p.is_empty()) {
            return true;
        }
        
        false
    }
    
    /// Get the "virtual" display_rows for backward compatibility
    /// Returns active_pane's display_rows if multi-pane, or full task_list if fallback
    pub fn display_rows(&self) -> &[DisplayRow] {
        if self.should_show_single_pane() {
            // In fallback mode, return panes[0].display_rows or empty vec if no panes
            if !self.panes.is_empty() {
                &self.panes[0].display_rows
            } else {
                &[]
            }
        } else {
            &self.panes[self.active_pane].display_rows
        }
    }
    
    /// Get mutable display_rows for updating (only valid in Normal mode)
    pub fn display_rows_mut(&mut self) -> &mut Vec<DisplayRow> {
        if self.should_show_single_pane() && !self.panes.is_empty() {
            &mut self.panes[0].display_rows
        } else if !self.panes.is_empty() {
            &mut self.panes[self.active_pane].display_rows
        } else {
            // Safety: ensure we always have at least one pane
            self.panes.push(Pane::new(0, "Tasks".to_string()));
            &mut self.panes[0].display_rows
        }
    }
    
    /// Ensure active_pane is always valid
    pub fn reconcile_active_pane(&mut self) {
        if self.panes.is_empty() {
            self.panes.push(Pane::new(0, "Tasks".to_string()));
            self.active_pane = 0;
        } else if self.active_pane >= self.panes.len() {
            self.active_pane = self.panes.len() - 1;
        }
    }
    
    /// Rebuild display rows for the active pane (or pane 0 in fallback)
    pub fn rebuild_visible_rows(&mut self) {
        let pane_idx = if self.should_show_single_pane() { 0 } else { self.active_pane };
        
        if pane_idx >= self.panes.len() {
            self.reconcile_active_pane();
            return;
        }
        
        let pane = &mut self.panes[pane_idx];
        let filter = Filter::default();  // TODO: Phase 25 uses pane.filter_query
        
        let mut rows: Vec<DisplayRow> = self
            .task_list
            .filter(&filter)
            .into_iter()
            .map(|(source_index, _task)| DisplayRow::Task(source_index))
            .collect();
        
        pane.display_rows = rows;
        
        // Reconcile selection
        if pane.selected >= pane.display_rows.len() && !pane.display_rows.is_empty() {
            pane.selected = pane.display_rows.len() - 1;
        } else if pane.display_rows.is_empty() {
            pane.selected = 0;
        }
    }
}

if app.should_show_single_pane() {
    // Single-pane fallback: render like pre-pane v1.3
    if !app.panes.is_empty() {
        let pane = &app.panes[0];
        TaskList::render(
            frame,
            list_area,
            pane.display_rows,
            pane.selected,
            &app.theme,
            matches!(app.mode, AppMode::Normal),
        );
    }
    // If panes.is_empty(), skip task list rendering (no tasks to show)
} else {
    // Multi-pane layout: render each pane side-by-side
    let pane_count = panes.len();
    let pane_constraints = vec![Constraint::Percentage(100 / pane_count as u16); pane_count];
    let pane_areas = Layout::default()
        .direction(Direction::Horizontal)
        .constraints(pane_constraints)
        .split(list_area);
    
    for (pane_idx, pane) in panes.iter().enumerate() {
        let pane_area = pane_areas[pane_idx];
        let is_active = pane_idx == app.active_pane;
        
        PaneList::render(
            frame,
            pane_area,
            pane,
            is_active,
            &app.theme,
            matches!(app.mode, AppMode::Normal),
        );
    }
}

#[cfg(test)]
mod fallback_tests {
    use todotxt_tui::app::{App, AppMode};
    use todotxt_tui::state::Pane;
    use std::path::PathBuf;
    
    #[test]
    fn test_single_pane_mode_with_empty_panes() {
        let mut app = App::new(PathBuf::from("test.txt")).unwrap();
        app.panes.clear();  // Remove all panes
        
        assert!(app.should_show_single_pane());
    }
    
    #[test]
    fn test_single_pane_mode_with_one_pane() {
        let app = App::new(PathBuf::from("test.txt")).unwrap();
        // Default: one pane
        
        assert!(app.should_show_single_pane());
    }
    
    #[test]
    fn test_single_pane_mode_with_all_empty() {
        let mut app = App::new(PathBuf::from("test.txt")).unwrap();
        app.panes.push(Pane::new(1, "Empty1".to_string()));
        app.panes.push(Pane::new(2, "Empty2".to_string()));
        // All panes are empty (no display_rows)
        
        assert!(app.should_show_single_pane());
    }
    
    #[test]
    fn test_multi_pane_mode_with_populated_panes() {
        let mut app = App::new(PathBuf::from("test.txt")).unwrap();
        app.panes.push(Pane::new(1, "Work".to_string()));
        
        // Add a task to pane 0
        app.panes[0].display_rows.push(DisplayRow::Task(0));
        
        assert!(!app.should_show_single_pane());
    }
    
    #[test]
    fn test_reconcile_empty_panes() {
        let mut app = App::new(PathBuf::from("test.txt")).unwrap();
        app.panes.clear();
        app.active_pane = 5;
        
        app.reconcile_active_pane();
        
        // Should have created one default pane
        assert_eq!(app.panes.len(), 1);
        assert_eq!(app.active_pane, 0);
    }
    
    #[test]
    fn test_reconcile_out_of_bounds_active_pane() {
        let mut app = App::new(PathBuf::from("test.txt")).unwrap();
        app.panes.push(Pane::new(1, "Work".to_string()));
        app.active_pane = 10;
        
        app.reconcile_active_pane();
        
        assert_eq!(app.active_pane, 1);  // Last valid index
    }
    
    #[test]
    fn test_display_rows_fallback() {
        let mut app = App::new(PathBuf::from("test.txt")).unwrap();
        app.panes[0].display_rows.push(DisplayRow::Task(0));
        
        // In single-pane mode, display_rows returns pane[0]
        let rows = app.display_rows();
        assert_eq!(rows.len(), 1);
    }
    
    #[test]
    fn test_display_rows_multi_pane() {
        let mut app = App::new(PathBuf::from("test.txt")).unwrap();
        app.panes.push(Pane::new(1, "Work".to_string()));
        
        // Add task to pane 0
        app.panes[0].display_rows.push(DisplayRow::Task(0));
        // Add task to pane 1
        app.panes[1].display_rows.push(DisplayRow::Task(1));
        app.panes[1].display_rows.push(DisplayRow::Task(2));
        
        app.active_pane = 1;
        
        // In multi-pane mode, display_rows returns active_pane
        let rows = app.display_rows();
        assert_eq!(rows.len(), 2);
    }
}
