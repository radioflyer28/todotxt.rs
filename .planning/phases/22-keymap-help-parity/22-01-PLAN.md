2. Add `parse_key_chord(s: &str) -> Option<(crossterm::event::KeyCode, crossterm::event::KeyModifiers)>` as a `pub(crate)` free function in config.rs:
   - Normalize input: `s.trim().to_ascii_lowercase()`
   - Split on `+` to separate modifiers from the final key token
   - Parse modifier tokens: `"ctrl"` → `KeyModifiers::CONTROL`, `"shift"` → `KeyModifiers::SHIFT`, `"alt"` → `KeyModifiers::ALT`; unrecognized modifier token → return `None`
   - Parse the final key token (last element after split):
     - Single char (e.g. `"n"`, `"?"`, `"0"`, `" "`) → `KeyCode::Char(c)`; `"space"` → `KeyCode::Char(' ')`
     - `"enter"` → `KeyCode::Enter`
     - `"esc"` → `KeyCode::Esc`
     - `"backspace"` → `KeyCode::Backspace`
     - `"delete"` → `KeyCode::Delete`
     - `"up"` → `KeyCode::Up`
     - `"down"` → `KeyCode::Down`
     - `"left"` → `KeyCode::Left`
     - `"right"` → `KeyCode::Right`
     - `"tab"` → `KeyCode::Tab`
     - `"f1"`–`"f12"` → `KeyCode::F(1)`–`KeyCode::F(12)`; strip `"f"` prefix, parse digit
     - Anything else → return `None`
   - Return `Some((key_code, modifiers))`

3. Add `fn default_keymap() -> HashMap<String, (crossterm::event::KeyCode, crossterm::event::KeyModifiers)>` (pub(crate)) that returns the 16 default bindings:
   ```
   "quit"          → (Char('q'), NONE)
   "add"           → (Char('n'), NONE)
   "edit"          → (Char('e'), NONE)      // 'u' stays as hardcoded alias in handle_normal_key
   "delete"        → (Char('d'), NONE)
   "bulk_delete"   → (Char('D'), NONE)
   "bulk_append"   → (Char('T'), NONE)
   "toggle_done"   → (Char('x'), NONE)
   "filter_open"   → (Char('f'), NONE)
   "filter_define" → (Char('F'), NONE)
   "filter_toggle" → (Char('f'), CONTROL)
   "sort_cycle"    → (Char('o'), NONE)
   "group_toggle"  → (Char('g'), NONE)
   "deferred_toggle" → (Char('h'), NONE)
   "theme_cycle"   → (Char('t'), NONE)
   "disjoint_select" → (Char('v'), NONE)
   "disjoint_mark" → (Char(' '), NONE)
   ```
   Use `KeyModifiers::NONE`, `KeyModifiers::CONTROL`, `KeyModifiers::SHIFT` as appropriate. Import `crossterm::event::{KeyCode, KeyModifiers}` at the top of config.rs.

4. Add `pub fn resolve_keymap(config: &TuiConfig) -> (HashMap<String, (crossterm::event::KeyCode, crossterm::event::KeyModifiers)>, Vec<String>)`:
   - Start with `let mut effective = default_keymap();`
   - Build the known action set from the default_keymap keys (`.keys().cloned().collect::<HashSet<_>>()`)
   - For each `(action, chord_str)` in `config.keymap`:
     - If action not in known set → push `format!("unknown action '{}' in [keymap] — ignored", action)` to warnings
     - Else if `parse_key_chord(chord_str)` is `None` → push `format!("invalid key chord '{}' for action '{}' in [keymap] — default used", chord_str, action)` to warnings
     - Else → `effective.insert(action.clone(), parsed_binding)`
   - Return `(effective, warnings)`
   - Conflict detection (two overrides landing on the same chord) is added in Plan 02 — do NOT add it here.
