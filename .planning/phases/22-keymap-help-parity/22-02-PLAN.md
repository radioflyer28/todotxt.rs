**config.rs — update resolve_keymap to add conflict detection (D-07):**

After applying all valid overrides into `effective`, perform conflict detection:
1. Build a reverse map: `chord → Vec<action_name>` by iterating over `effective`
2. For any chord that maps to more than one action: add both action names to warnings with message format:
   `format!("conflict: '{}' and '{}' both map to '{}' — both reverted to defaults", action_a, action_b, chord_str)`
   Then revert both actions to their default bindings (look up in `default_keymap()`).
3. Note: this catches cases where the user configured action A to chord X AND action B to chord X, or where the user configured action A to chord X and chord X was ALREADY the default binding of action B (and B was not overridden).

Exact conflict detection algorithm:
```rust
// After applying overrides, build chord → [action] reverse map
let mut chord_to_actions: HashMap<(KeyCode, KeyModifiers), Vec<String>> = HashMap::new();
for (action, binding) in &effective {
    chord_to_actions.entry(*binding).or_default().push(action.clone());
}
let defaults = default_keymap();
for (chord, actions) in &chord_to_actions {
    if actions.len() > 1 {
        // Describe chord as "KeyCode:Modifiers" for the warning message
        let chord_desc = format!("{:?}+{:?}", chord.0, chord.1);
        warnings.push(format!(
            "conflict: actions [{}] all bound to {} — all reverted to defaults",
            actions.join(", "), chord_desc
        ));
        for action in actions {
            if let Some(default_binding) = defaults.get(action.as_str()) {
                effective.insert(action.clone(), *default_binding);
            }
        }
    }
}
```

   ```rust
   fn render_keymap_errors_overlay(&self, frame: &mut Frame, area: ratatui::layout::Rect) {
       use ratatui::layout::{Constraint, Layout, Direction};
       use ratatui::widgets::{Block, Borders, List, ListItem, Clear};
       use ratatui::style::Style;

       let warning_count = self.keymap_warnings.len();
       // Panel height: border (2) + warning lines, capped at 80% of terminal height
       let panel_height = (warning_count as u16 + 2).min((area.height * 4) / 5).max(4);
       // Center vertically
       let vert = Layout::default()
           .direction(Direction::Vertical)
           .constraints([
               Constraint::Min(0),
               Constraint::Length(panel_height),
               Constraint::Min(0),
           ])
           .split(area);
       let popup_area = vert[1];

       // Clear the background behind the overlay
       frame.render_widget(Clear, popup_area);

       let outer = Block::default()
           .title(" Keymap Warnings — Esc/q: close ")
           .borders(Borders::ALL);
       let inner = outer.inner(popup_area);
       frame.render_widget(outer, popup_area);

       let items: Vec<ListItem> = self
           .keymap_warnings
           .iter()
           .map(|w| ListItem::new(format!("  ⚠ {}", w)))
           .collect();
       frame.render_widget(List::new(items), inner);
   }
   ```
