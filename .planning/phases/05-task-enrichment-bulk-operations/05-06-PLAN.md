```rust
#[test]
fn pri_sets_priority_on_single_task() {
    let fixture = TestFixture::new();
    fixture.write_todo("Learn Rust");
    fixture.write_todo("Write code");
    
    fixture.assert_cmd("pri")
        .arg("1")
        .arg("A")
        .success()
        .stdout(contains("Set priority to (A)"));
    
    fixture.assert_file_contains("todo.txt", "(A) Learn Rust");
}

#[test]
fn due_weekday_fixed_date() {
    // Use fixed test date: 2026-04-16 (Thursday)
    // Test: `due 1 friday` should set to 2026-04-17
    let fixture = TestFixture::new_with_date(NaiveDate::from_ymd_opt(2026, 4, 16).unwrap());
    fixture.write_todo("Task");
    
    fixture.assert_cmd("due")
        .arg("1")
        .arg("friday")
        .success();
    
    fixture.assert_file_contains("todo.txt", "due:2026-04-17");
}

#[test]
fn archive_atomicity() {
    let fixture = TestFixture::new();
    fixture.write_todo("x 2026-01-01 Completed task");
    fixture.write_todo("Learn Rust");
    
    fixture.assert_cmd("archive").success();
    
    // Verify todo.txt has only incomplete
    fixture.assert_file_contains("todo.txt", "Learn Rust");
    fixture.assert_file_not_contains("todo.txt", "x 2026");
    
    // Verify done.txt has completed
    fixture.assert_file_contains("done.txt", "x 2026");
}
```
