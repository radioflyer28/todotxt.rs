```rust
use std::fs;
use std::path::PathBuf;

pub fn run_archive(
    list: TaskList,
    cfg: &Config,
    renderer: &Renderer,
) -> Result<(), CliError> {
    // 1. Filter completed vs incomplete tasks
    let completed: Vec<_> = list.tasks()
        .filter(|t| t.is_completed())
        .cloned()
        .collect();
    
    let incomplete: Vec<_> = list.tasks()
        .filter(|t| !t.is_completed())
        .cloned()
        .collect();
    
    // 2. Resolve done.txt path (from cfg.done_file or sibling)
    let done_file_path = cfg.done_file.as_ref()
        .map(|p| p.clone())
        .or_else(|| {
            list.path.parent().map(|p| p.join("done.txt"))
        })
        .ok_or_else(|| CliError::Other("Cannot resolve done.txt path".to_string()))?;
    
    // 3. Read existing done.txt (if exists)
    let mut done_tasks = if done_file_path.exists() {
        TaskList::load(&done_file_path)?
    } else {
        TaskList::new()
    };
    
    // 4. Append completed tasks to done_tasks
    for task in &completed {
        done_tasks.add(task.clone());
    }
    
    // 5. Atomic write: update both files atomically
    //    (TaskList::save uses temp-file + rename internally)
    // 5a. Write new done.txt
    done_tasks.save(&done_file_path)?;
    
    // 5b. Write new todo.txt (with only incomplete tasks)
    let mut updated_list = TaskList::new();
    for task in &incomplete {
        updated_list.add(task.clone());
    }
    updated_list.save(&list.path)?;
    
    // 6. Report
    let count = completed.len();
    let msg = if cfg.json_output {
        // Return JSON envelope
        json_success_with_count(count)
    } else {
        format!("Archived {} completed task{}", count, if count == 1 { "" } else { "s" })
    };
    
    renderer.print_info(&msg);
    Ok(())
}

fn json_success_with_count(count: usize) -> String {
    format!(
        r#"{{"schema_version":1,"status":"ok","count":{}}}"#,
        count
    )
}
```

Design notes:
- Idempotent: running archive twice is safe (count will be 0 on second run)
- Atomic: both todo.txt and done.txt written before function returns
- Output: "Archived N completed tasks" to stderr (per print_info), exit 0
- JSON mode: return `{"schema_version":1,"status":"ok","count":N}`
- Creates done.txt if absent

```rust
pub fn run_del_done(
    list: TaskList,
    cfg: &Config,
    renderer: &Renderer,
) -> Result<(), CliError> {
    // 1. Filter for incomplete tasks only
    let mut incomplete = TaskList::new();
    let mut deleted_count = 0;
    
    for task in list.tasks() {
        if !task.is_completed() {
            incomplete.add(task.clone());
        } else {
            deleted_count += 1;
        }
    }
    
    // 2. Write updated list (atomic via TaskList::save)
    incomplete.save()?;
    
    // 3. Report
    let msg = if cfg.json_output {
        json_success_with_count(deleted_count)
    } else {
        format!("Deleted {} completed task{}", deleted_count, if deleted_count == 1 { "" } else { "s" })
    };
    
    renderer.print_info(&msg);
    Ok(())
}

fn json_success_with_count(count: usize) -> String {
    format!(
        r#"{{"schema_version":1,"status":"ok","count":{}}}"#,
        count
    )
}
```

Design notes:
- Idempotent: count is 0 if no completed tasks
- In-place: overwrites todo.txt with incomplete tasks only
- Output: "Deleted N completed tasks" to stderr
- JSON mode: return `{"schema_version":1,"status":"ok","count":N}`
- Atomic: TaskList::save handles temp-file + rename
