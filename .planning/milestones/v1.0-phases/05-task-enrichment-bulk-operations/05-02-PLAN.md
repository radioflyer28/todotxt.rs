```rust
use chrono::{Datelike, Local, NaiveDate, Weekday};

pub fn parse_date_input(input: &str, today: NaiveDate) -> Result<NaiveDate, String> {
    let trimmed = input.trim().to_lowercase();
    
    match trimmed.as_str() {
        "today" => Ok(today),
        "tomorrow" => Ok(today + chrono::Duration::days(1)),
        weekday_str => {
            // Parse weekday names (monday, tuesday, ..., sunday)
            // Return next occurrence of that weekday
            // If today IS that weekday, return next week (7 days later)
            let weekday = match_weekday(weekday_str)?;
            Ok(next_weekday(today, weekday))
        }
        _ => {
            // Try YYYY-MM-DD format
            NaiveDate::parse_from_str(&trimmed, "%Y-%m-%d")
                .map_err(|_| "Invalid date format. Use: today, tomorrow, weekday name, or YYYY-MM-DD".to_string())
        }
    }
}

fn match_weekday(s: &str) -> Result<Weekday, String> {
    match s {
        "monday" => Ok(Weekday::Mon),
        "tuesday" => Ok(Weekday::Tue),
        "wednesday" => Ok(Weekday::Wed),
        "thursday" => Ok(Weekday::Thu),
        "friday" => Ok(Weekday::Fri),
        "saturday" => Ok(Weekday::Sat),
        "sunday" => Ok(Weekday::Sun),
        _ => Err(format!("Unknown weekday: {}", s)),
    }
}

fn next_weekday(from: NaiveDate, target: Weekday) -> NaiveDate {
    let mut current = from;
    loop {
        current = current + chrono::Duration::days(1);
        if current.weekday() == target {
            return current;
        }
    }
}
```

This implementation:
- Returns Result<NaiveDate, String> for error handling
- Accepts today/tomorrow as constants
- Accepts weekday names (case-insensitive)
- Returns next occurrence of weekday (if today is Monday and user says "monday", returns next Monday, 7 days out)
- Accepts YYYY-MM-DD
- Rejects +N shorthand (per D-03, that's only for postpone N)
