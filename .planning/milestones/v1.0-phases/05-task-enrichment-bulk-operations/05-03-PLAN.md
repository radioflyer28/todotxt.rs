```rust
pub fn run_pri(
    mut list: TaskList,
    ids: Vec<u32>,
    priority: char,
    cfg: &Config,
    renderer: &Renderer,
) -> Result<(), CliError> {
    // 1. Validate priority letter (A-Z, case-insensitive)
    let priority_upper = priority.to_ascii_uppercase();
    if !(priority_upper >= 'A' && priority_upper <= 'Z') {
        return Err(CliError::ValidationError(
            format!("Invalid priority: {}. Use A-Z.", priority),
            ExitCode::ValidationError,
        ));
    }
    
    // 2. Validate ALL IDs before mutating (fail-fast per D-01)
    if ids.is_empty() {
        return Err(CliError::ValidationError(
            "No task IDs specified.".to_string(),
            ExitCode::ValidationError,
        ));
    }
    for &id in &ids {
        validate_id(id, list.len())?;
    }
    
    // 3. Sort descending, dedup (prevent index shift during iteration)
    let mut sorted_ids = ids;
    sorted_ids.sort_by(|a, b| b.cmp(a));
    sorted_ids.dedup();
    
    // 4. Update tasks
    for &id in &sorted_ids {
        let idx = (id - 1) as usize;
        let task = list.get(idx).ok_or(CliError::NotFound)?;
        let updated = task.with_priority(priority_upper);
        list.update(idx, updated.clone())?;
        renderer.print_write_result(
            &format!("Set priority to ({})", priority_upper),
            id as usize,
            &updated,
        );
    }
    
    list.save()?;
    Ok(())
}

pub fn run_depri(
    mut list: TaskList,
    ids: Vec<u32>,
    cfg: &Config,
    renderer: &Renderer,
) -> Result<(), CliError> {
    // 1. Validate ALL IDs before mutating
    if ids.is_empty() {
        return Err(CliError::ValidationError(
            "No task IDs specified.".to_string(),
            ExitCode::ValidationError,
        ));
    }
    for &id in &ids {
        validate_id(id, list.len())?;
    }
    
    // 2. Sort descending, dedup
    let mut sorted_ids = ids;
    sorted_ids.sort_by(|a, b| b.cmp(a));
    sorted_ids.dedup();
    
    // 3. Remove priority from tasks
    for &id in &sorted_ids {
        let idx = (id - 1) as usize;
        let task = list.get(idx).ok_or(CliError::NotFound)?;
        let updated = task.without_priority(); // Method may be new or reuse with_priority(None)
        list.update(idx, updated.clone())?;
        renderer.print_write_result("Removed priority", id as usize, &updated);
    }
    
    list.save()?;
    Ok(())
}

fn validate_id(id: u32, list_len: usize) -> Result<(), CliError> {
    if id == 0 || id as usize > list_len {
        Err(CliError::NotFound)
    } else {
        Ok(())
    }
}
```
