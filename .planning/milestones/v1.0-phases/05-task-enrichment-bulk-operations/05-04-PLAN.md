```rust
use crate::date::parse_date_input;
use chrono::Local;

pub fn run_due(
    mut list: TaskList,
    id: u32,
    date_str: &str,
    cfg: &Config,
    renderer: &Renderer,
) -> Result<(), CliError> {
    // 1. Validate ID
    validate_id(id, list.len())?;
    
    // 2. Parse date
    let today = Local::now().naive_utc().date();
    let due_date = parse_date_input(date_str, today)
        .map_err(|e| CliError::ValidationError(e, ExitCode::ValidationError))?;
    
    // 3. Update task with due:YYYY-MM-DD tag
    let idx = (id - 1) as usize;
    let task = list.get(idx).ok_or(CliError::NotFound)?;
    let due_tag = format!("due:{}", due_date.format("%Y-%m-%d"));
    let updated = task.with_tag("due", &due_tag); // or task.set_due_date(due_date)
    list.update(idx, updated.clone())?;
    renderer.print_write_result(&format!("Set due date to {}", due_date), id as usize, &updated);
    
    list.save()?;
    Ok(())
}

pub fn run_postpone(
    mut list: TaskList,
    id: u32,
    days: u32,
    cfg: &Config,
    renderer: &Renderer,
) -> Result<(), CliError> {
    // 1. Validate ID
    validate_id(id, list.len())?;
    
    // 2. Get task and extract existing due date
    let idx = (id - 1) as usize;
    let task = list.get(idx).ok_or(CliError::NotFound)?;
    
    // Extract due:YYYY-MM-DD from tags
    let current_due = task.tags
        .get("due")
        .and_then(|due_tag| {
            NaiveDate::parse_from_str(due_tag, "due:%Y-%m-%d")
                .or_else(|_| NaiveDate::parse_from_str(due_tag, "%Y-%m-%d"))
                .ok()
        })
        .ok_or_else(|| {
            CliError::ValidationError(
                format!("Task {} has no due date to postpone", id),
                ExitCode::ValidationError,
            )
        })?;
    
    // 3. Add days
    let new_due = current_due + chrono::Duration::days(days as i64);
    let due_tag = format!("due:{}", new_due.format("%Y-%m-%d"));
    let updated = task.with_tag("due", &due_tag);
    list.update(idx, updated.clone())?;
    renderer.print_write_result(
        &format!("Postponed due date {} days to {}", days, new_due),
        id as usize,
        &updated,
    );
    
    list.save()?;
    Ok(())
}

fn validate_id(id: u32, list_len: usize) -> Result<(), CliError> {
    if id == 0 || id as usize > list_len {
        Err(CliError::NotFound)
    } else {
        Ok(())
    }
}
```

Design notes:
- run_due uses parse_date_input to accept all ROADMAP formats
- run_postpone extracts existing due date from task tags, adds days, writes back
- Both use renderer.print_write_result for output
- Exit code 2 for validation errors (invalid date, missing due date on postpone)
- Exit code 1 for ID not found
