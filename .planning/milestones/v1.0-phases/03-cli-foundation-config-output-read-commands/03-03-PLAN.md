```rust
use clap::CommandFactory;
use clap_complete::{generate, Shell};
use crate::cli::Cli;

/// Generate shell completions and write to stdout. Used by `todotxt completions <shell>`.
///
/// The Shell enum from clap_complete supports: Bash, Zsh, Fish, PowerShell, Elvish.
pub fn run(shell: Shell) {
    let mut cmd = Cli::command();
    generate(shell, &mut cmd, "todotxt", &mut std::io::stdout());
}
```

This function has no return type — it writes directly to stdout and cannot fail. The `Shell` argument is parsed by clap from the CLI (variant of `clap_complete::Shell` re-exported via `clap_complete::aot`).

Note: The `Commands::Completions { shell }` arm in `main.rs` calls `commands::completions::run(*shell)` — no `?` needed since it cannot error.

```rust
use assert_cmd::Command;
use assert_fs::{fixture::ChildPath, prelude::*, TempDir};

pub const SAMPLE_TODO: &str = "\
(A) Buy milk +groceries @home due:2099-01-01\n\
(B) Send report +work @office\n\
x 2024-01-01 Done task +work\n\
Call dentist @personal\n\
";

/// Create a TempDir with a todo.txt and config.toml pointing to it.
pub struct TestFixture {
    pub dir: TempDir,
    pub todo: ChildPath,
    pub config: ChildPath,
}

impl TestFixture {
    pub fn new() -> Self {
        Self::with_content(SAMPLE_TODO)
    }

    pub fn with_content(content: &str) -> Self {
        let dir = TempDir::new().expect("TempDir");
        let todo = dir.child("todo.txt");
        todo.write_str(content).expect("write todo.txt");
        let config = dir.child("config.toml");
        config
            .write_str(&format!("todo_file = {:?}\n", todo.path()))
            .expect("write config.toml");
        Self { dir, todo, config }
    }

    pub fn cmd(&self) -> Command {
        let mut c = Command::cargo_bin("todotxt").expect("todotxt binary");
        c.arg("--config").arg(self.config.path());
        c
    }
}
```

```rust
#[test]
fn list_empty_filter_exits_zero() {
    // P-10: no matching tasks = exit 0, not exit 1
    let fx = TestFixture::new();
    fx.cmd()
        .arg("list")
        .arg("xyznonexistenttoken12345")
        .assert()
        .success();  // exit 0
}

#[test]
fn list_positional_filter_narrows_results() {
    let fx = TestFixture::new();
    fx.cmd()
        .arg("list")
        .arg("report")
        .assert()
        .success()
        .stdout(predicate::str::contains("report"))
        .stdout(predicate::str::contains("Buy milk").not());
}

#[test]
fn list_json_envelope() {
    let fx = TestFixture::new();
    fx.cmd()
        .arg("--json")
        .arg("list")
        .assert()
        .success()
        .stdout(predicate::str::contains("\"schema_version\":1"))
        .stdout(predicate::str::contains("\"data\""));

#[test]
fn list_no_color_no_ansi() {
    let fx = TestFixture::new();
    let output = fx.cmd().arg("--no-color").arg("list").output().unwrap();
```
