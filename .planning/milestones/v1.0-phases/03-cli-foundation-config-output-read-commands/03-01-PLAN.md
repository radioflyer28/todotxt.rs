```rust
use anyhow::{anyhow, Context, Result};
use directories::ProjectDirs;
use serde::{Deserialize, Serialize};
use std::collections::HashMap;
use std::path::{Path, PathBuf};
use todotxt_core::resolve_config_path;

/// Named filter preset stored in config under `[presets.name]`.
#[derive(Debug, Deserialize, Serialize, Default, Clone)]
pub struct PresetConfig {
    pub filter: Option<String>,
}

/// Application configuration loaded from TOML.
#[derive(Debug, Deserialize, Serialize, Default)]
pub struct Config {
    /// Path to the user's todo.txt file. Required after first run.
    pub todo_file: Option<PathBuf>,
    /// Named filter presets. Max 9 per CFG-02.
    #[serde(default)]
    pub presets: HashMap<String, PresetConfig>,
}

impl Config {
    /// Returns the platform-appropriate config directory path.
    ///
    /// - Linux:   ~/.config/todotxt/config.toml
    /// - Windows: %APPDATA%\todotxt\config.toml
    /// - macOS:   ~/Library/Application Support/todotxt/config.toml
    pub fn default_path() -> Option<PathBuf> {
        ProjectDirs::from("", "", "todotxt")
            .map(|dirs| dirs.config_dir().join("config.toml"))
    }

    /// Resolves the config path, applying portable mode:
    /// if config.toml exists beside the binary, use that path instead.
    pub fn resolve_path(platform_path: &Path) -> PathBuf {
        let binary_dir = std::env::current_exe()
            .ok()
            .and_then(|p| p.parent().map(|d| d.to_path_buf()))
            .unwrap_or_else(|| PathBuf::from("."));
        resolve_config_path(&binary_dir, platform_path.parent().unwrap_or(platform_path))
            .join("config.toml")
    }

    /// Load config from `path`. If the file does not exist, auto-create it
    /// with defaults (todo_file = ~/todo.txt) and return the default config.
    ///
    /// # Errors
    /// Returns Err if the file exists but is malformed TOML, or if filesystem
    /// operations fail.
    pub fn load_or_create(path: &Path) -> Result<Self> {
        if path.exists() {
            let content = std::fs::read_to_string(path)
                .with_context(|| format!("reading config: {}", path.display()))?;
            toml::from_str(&content)
                .with_context(|| format!("parsing config: {}", path.display()))
        } else {
            // Auto-create with default todo_file = ~/todo.txt (D-01, D-02)
            let home_todo = dirs_next_home().map(|h| h.join("todo.txt"));
            let default = Config {
                todo_file: home_todo,
                presets: HashMap::new(),
            };
            let toml_str = toml::to_string_pretty(&default)
                .context("serializing default config")?;
            if let Some(parent) = path.parent() {
                std::fs::create_dir_all(parent)  // P-08: always create dirs first
                    .with_context(|| format!("creating config dir: {}", parent.display()))?;
            }
            std::fs::write(path, &toml_str)
                .with_context(|| format!("writing default config: {}", path.display()))?;
            Ok(default)
        }
    }

    /// Returns the resolved todo.txt path, or an error if not configured (D-03).
    pub fn resolve_todo_file(&self) -> Result<PathBuf> {
        self.todo_file.clone().ok_or_else(|| {
            anyhow!(
                "todo_file not set in config. \
                 Add `todo_file = \"/path/to/todo.txt\"` to your config or use --todo-file"
            )
        })
    }
}

fn dirs_next_home() -> Option<PathBuf> {
    directories::BaseDirs::new().map(|b| b.home_dir().to_path_buf())
}
```

use comfy_table::presets::NOTHING;
use comfy_table::{Cell, ContentArrangement, Table};
use owo_colors::{set_override, OwoColorize};
use serde::Serialize;
use serde_json::json;
use todotxt_core::Task;

/// Initialize global color state. Must be called before any output.
/// Checks both `--no-color` flag and `NO_COLOR` environment variable (D-07 / P-02).
pub fn init_color(no_color: bool) {
    let env_no_color = std::env::var_os("NO_COLOR").is_some();
    if no_color || env_no_color {
        set_override(false);
    }
}

/// Central renderer — encapsulates output mode (JSON vs human) and quiet state.
pub struct Renderer {
    pub json: bool,
    pub quiet: bool,
}

impl Renderer {
    pub fn new(json: bool, quiet: bool) -> Self {
        Self { json, quiet }
    }

    /// Write informational message to stderr. Suppressed by --quiet (D-14).
    pub fn info(&self, msg: &str) {
        if !self.quiet {
            eprintln!("{}", msg);
        }
    }

    /// Write error message to stderr. NEVER suppressed by --quiet (D-14).
    pub fn error(&self, msg: &str) {
        eprintln!("error: {}", msg);
    }

    /// Print task list in human table or JSON format.
    pub fn print_tasks(&self, tasks: &[(usize, &Task)]) {
        if self.json {
            let dtos: Vec<_> = tasks.iter().map(|(idx, t)| task_to_dto(*idx, t)).collect();
            println!("{}", json_success(dtos));
        } else {
            let table = build_task_table(tasks);
            println!("{table}");
        }
    }

    /// Print a single task (for `show` command).
    pub fn print_task(&self, idx: usize, task: &Task) {
        if self.json {
            println!("{}", json_success(task_to_dto(idx, task)));
        } else {
            println!("{}", task.to_raw());  // D-13: raw line only
        }
    }

    /// Print lines of plain text (projects, contexts — one per line).
    pub fn print_lines(&self, lines: &[String]) {
        if self.json {
            println!("{}", json_success(lines));
        } else {
            for line in lines {
                println!("{}", line);
            }
        }
    }

    /// Print a generic serializable value as JSON or formatted key-value pairs.
    pub fn print_stats(&self, stats: &Stats) {
        if self.json {
            println!("{}", json_success(stats));
        } else {
            println!("Total:      {}", stats.total);
            println!("Complete:   {}", stats.complete);
            println!("Incomplete: {}", stats.incomplete);
            println!("Due today:  {}", stats.due_today);
            println!("Overdue:    {}", stats.overdue);
        }
    }
}

#[derive(Serialize)]
pub struct Stats {
    pub total: usize,
    pub complete: usize,
    pub incomplete: usize,
    pub due_today: usize,
    pub overdue: usize,
}

/// JSON success envelope: `{"schema_version":1,"data":<T>}`
pub fn json_success<T: Serialize>(data: T) -> String {
    serde_json::to_string(&json!({
        "schema_version": 1,
        "data": data
    }))
    .expect("JSON serialization is infallible for valid types")
}

/// JSON error envelope: `{"schema_version":1,"error":"<msg>"}` (P-09: goes to stdout when --json)
pub fn json_error(msg: &str) -> String {
    serde_json::to_string(&json!({
        "schema_version": 1,
        "error": msg
    }))
    .expect("JSON serialization is infallible")
}

fn build_task_table(tasks: &[(usize, &Task)]) -> Table {
    let mut table = Table::new();
    table
        .load_preset(NOTHING)
        .set_content_arrangement(ContentArrangement::Dynamic)
        .set_header(vec!["ID", "PRI", "Task"]);  // D-07: header row only

    for (idx, task) in tasks {
        let id = idx + 1;  // 1-based display ID (Phase 1 decision)
        let pri_badge = format_priority(task.priority);
        table.add_row(vec![
            Cell::new(id.to_string()),
            Cell::new(pri_badge),
            Cell::new(task.to_raw()),
        ]);
    }
    table
}

fn format_priority(priority: Option<char>) -> String {
    match priority {
        Some(p) => {
            let badge = format!("({})", p);
            // D-06: priority badge colored by level; no task text colorization
            match p {
                'A' => badge.red().to_string(),
                'B' => badge.yellow().to_string(),
                'C' => badge.green().to_string(),
                _ => badge,
            }
        }
        None => String::new(),
    }
}

#[derive(Serialize)]
struct TaskDto {
    id: usize,
    raw: String,
    completed: bool,
    priority: Option<char>,
    projects: Vec<String>,
    contexts: Vec<String>,
    due_date: Option<String>,
}

fn task_to_dto(idx: usize, task: &Task) -> TaskDto {
    TaskDto {
        id: idx + 1,
        raw: task.to_raw().to_string(),
        completed: task.completed,
        priority: task.priority,
        projects: task.projects.clone(),
        contexts: task.contexts.clone(),
        due_date: task.due_date.map(|d| d.format("%Y-%m-%d").to_string()),
    }
}

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Config file → app | TOML content is user-authored; malformed TOML is caught by toml::from_str |
| Filesystem paths in config | todo_file path from config is user-controlled |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-01 | Tampering | config.rs | mitigate | Config is parsed via serde/toml — no eval, no code execution. Malformed TOML returns Err, not panic. |
| T-03-02 | Information Disclosure | output.rs json_error | mitigate | json_error() takes a pre-formatted message string — callers must NOT pass raw Rust error chains with internal paths; use `.to_string()` on anyhow errors which are already user-facing messages. |
| T-03-03 | Tampering | config.rs create_dir_all | accept | Config dir creation follows OS-standard path from `directories` crate; no path traversal risk since the path is fully constructed, not user-specified. |
</threat_model>
