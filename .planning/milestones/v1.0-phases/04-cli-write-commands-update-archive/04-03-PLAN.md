```rust
use crate::{output::Renderer, CliError};
use std::path::Path;
use todotxt_core::TaskList;

/// Mark one or more tasks complete (`do`). Per D-02: multi-ID, descending order.
/// Per D-03: already-completed tasks are skipped (idempotent, exit 0).
pub fn run_do(todo_path: &Path, ids: &[usize], renderer: &Renderer) -> Result<(), CliError> {
    if ids.is_empty() {
        return Err(CliError::Other(anyhow::anyhow!("at least one task ID required")));
    }

    let mut list = TaskList::load(todo_path)?;

    // Validate all IDs and collect 0-based indices.
    let mut indices: Vec<usize> = Vec::with_capacity(ids.len());
    for &id in ids {
        let idx = validate_id(id, list.len())?;
        indices.push(idx);
    }

    // Descending order — consistent with D-02 (no-op for updates but required by spec).
    indices.sort_unstable_by(|a, b| b.cmp(a));
    indices.dedup();

    for idx in indices {
        let task = list.tasks()[idx].clone();
        if task.completed {
            // D-03: already done — idempotent, print info to stderr, skip.
            eprintln!("info: task {} is already completed, skipping.", idx + 1);
            continue;
        }
        let updated = task.with_completed(true);
        list.update(idx, updated.clone())?;
        renderer.print_write_result(
            &format!("Completed task #{}.", idx + 1),
            idx,
            &updated,
        );
    }

    Ok(())
}

/// Unmark one or more completed tasks (`undo`). Per D-02: multi-ID, descending order.
/// Per D-03: already-incomplete tasks are skipped (idempotent, exit 0).
pub fn run_undo(todo_path: &Path, ids: &[usize], renderer: &Renderer) -> Result<(), CliError> {
    if ids.is_empty() {
        return Err(CliError::Other(anyhow::anyhow!("at least one task ID required")));
    }

    let mut list = TaskList::load(todo_path)?;

    let mut indices: Vec<usize> = Vec::with_capacity(ids.len());
    for &id in ids {
        let idx = validate_id(id, list.len())?;
        indices.push(idx);
    }

    indices.sort_unstable_by(|a, b| b.cmp(a));
    indices.dedup();

    for idx in indices {
        let task = list.tasks()[idx].clone();
        if !task.completed {
            // D-03: already incomplete — idempotent, print info to stderr, skip.
            eprintln!("info: task {} is already incomplete, skipping.", idx + 1);
            continue;
        }
        let updated = task.with_completed(false);
        list.update(idx, updated.clone())?;
        renderer.print_write_result(
            &format!("Undid task #{}.", idx + 1),
            idx,
            &updated,
        );
    }

    Ok(())
}

/// Convert a 1-based display ID to a 0-based index, validating it is in range.
fn validate_id(id: usize, list_len: usize) -> Result<usize, CliError> {
    if id == 0 {
        return Err(CliError::NotFound(format!(
            "task ID 0 is invalid (IDs start at 1)"
        )));
    }
    let idx = id - 1;
    if idx >= list_len {
        return Err(CliError::NotFound(format!(
            "task {} not found (list has {} tasks)",
            id, list_len
        )));
    }
    Ok(idx)
}
```

```rust
use crate::{output::Renderer, CliError};
use std::path::Path;
use todotxt_core::TaskList;

/// Delete one or more tasks by 1-based ID. Per D-04: validate all before deleting any.
/// Per D-02: process in descending order to prevent index shift during multi-delete.
/// Per D-08: no confirmation prompt.
pub fn run(todo_path: &Path, ids: &[usize], renderer: &Renderer) -> Result<(), CliError> {
    if ids.is_empty() {
        return Err(CliError::Other(anyhow::anyhow!("at least one task ID required")));
    }

    let mut list = TaskList::load(todo_path)?;

    // Step 1: Validate ALL IDs before deleting any (fail-fast, D-04).
    let mut indices: Vec<usize> = Vec::with_capacity(ids.len());
    for &id in ids {
        let idx = validate_id(id, list.len())?;
        indices.push(idx);
    }

    // Step 2: Sort descending + dedup — deletion in descending order prevents index shift.
    indices.sort_unstable_by(|a, b| b.cmp(a));
    indices.dedup();

    // Step 3: Print then delete (D-04: print deleted task to stdout BEFORE removal).
    for idx in indices {
        let task = list.tasks()[idx].clone();
        renderer.print_write_result(
            &format!("Deleted task #{}.", idx + 1),
            idx,
            &task,
        );
        list.delete(idx)?;
    }

    Ok(())
}

fn validate_id(id: usize, list_len: usize) -> Result<usize, CliError> {
    if id == 0 {
        return Err(CliError::NotFound(format!(
            "task ID 0 is invalid (IDs start at 1)"
        )));
    }
    let idx = id - 1;
    if idx >= list_len {
        return Err(CliError::NotFound(format!(
            "task {} not found (list has {} tasks)",
            id, list_len
        )));
    }
    Ok(idx)
}
```

Note: `list.delete(idx)?` atomically writes after each deletion. This means the file is always in a consistent state even if the process is interrupted mid-multi-delete.
